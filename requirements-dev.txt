# 테스트 스위트를 돌리는 데 필요한 것 전부.
#
# 이 목록이 없어서 실측으로 하나씩 찾아 넣은 적이 있다 — pytest 만 넣고
# 돌리면 `fitz` 없다는 collection 에러 3개(`test_hwp_splitter`·
# `test_rename_reports`·`test_split_quiz_etc`)가 나면서 **스위트 전체가
# 멈춘다.** 일부 실패가 아니라 한 줄도 못 돈다. 그래서 넷을 한 곳에 적는다.
#
# 아래 버전은 2026-09-06 에 python 3.14.6 / macOS 에서 477 tests +
# 231 subtests 전량 통과를 확인한 조합이다. 하한으로 적어 두되, 올렸다가
# 깨지면 이 줄을 근거로 되돌린다.
#
# 설치 — Homebrew 파이썬은 PEP 668 로 시스템 설치를 막는다. `pip install`
# 이 externally-managed 오류를 내면 `--break-system-packages` 를 쓰지 말고
# (Homebrew 설치를 망가뜨릴 수 있다) venv 를 만든다:
#
#     python3 -m venv .venv
#     .venv/bin/pip install -r requirements-dev.txt
#     .venv/bin/python -m pytest tests/ -q
#
# 런타임 의존성은 따로 없다 — `scripts/lab/` 은 표준 라이브러리만 쓰고
# PDF 변환은 `과목정보.md` 의 `chrome` 목록(macOS·Windows) 중 이 머신에
# 있는 경로로 헤드리스 Chrome 을 부른다.
# 아래 넷은 **테스트와 검증 스크립트 전용**이다.

pytest>=9.1.1

# pypdf — 체크리스트 PDF 의 A4 실측(210×297mm)과 페이지별 이미지 개수 확인.
# 브레드보드 그림이 깨진 채 PDF 로 굳는 사고를 잡아낸 것이 이 확인이다.
pypdf>=6.17.0

# pymupdf(`import fitz`) — hwp/PDF 분할·이름변경 스크립트의 테스트.
pymupdf>=1.28.2

# pillow — 교안 렌더 PNG 를 크롭·확대해 회로도를 vision 판독할 때.
pillow>=12.3.0
