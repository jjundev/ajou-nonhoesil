library ieee;  -- library 선언, ieee 라이브러리 사용함.								
use ieee.std_logic_1164.all;
use ieee.std_logic_unsigned.all;
use ieee.std_logic_1164.all;

entity switch_segment is    --entity명이 switch_segment.
 port (
-----------------------------
-- Pins declaration.
-----------------------------
-- on board clock source, PIN_R8
  clock_50m :in std_logic; -- 50MHz clock, 내장 clock
  pb : in std_logic_vector(16 downto 1);     --버튼 스위치 입력 선언
  fnd_s : out std_logic_vector(3 downto 0); --segment select 출력 선언
  fnd_d : out std_logic_vector(7 downto 0)); --segment data bus 출력 선언
end switch_segment;
 
architecture body_switch_segment of switch_segment is --architecture 선언

--상태가 sw_idle, sw_start, sw_s1, sw_s2, sw_s3, sw_s4, sw_s5, sw_s6,sw_save, sw_cancel을 갖는 state선언
type state is (sw_idle, sw_start, sw_s1, sw_s2, sw_s3, sw_s4, sw_s5, sw_s6, sw_save, sw_save1, sw_save2, sw_save3, sw_save4, sw_save5, sw_save6, sw_cancel);
signal sw_status : state:=sw_idle; --sw_status 신호선언 초기값에 sw_idle 대입
signal init_counter : std_logic_vector(31 downto 0); --init_counter신호선언 clock 분주기
signal sw_clk : std_logic;    --sw_clk신호선언, 버튼 감지용 clock
signal sw_toggle : std_logic:='0';
signal npb : std_logic_vector(16 downto 1):=X"0000"; --npb신호 선언, 스위치의 not값
signal pb_1st : std_logic_vector(16 downto 1):=X"0000"; --pb_1st신호선언, 스위치 처음 감지된값
signal pb_2nd : std_logic_vector(16 downto 1):=X"0000"; --pb_2nd신호선언, 스위치 두번째 감지된값
--비밀번호 4자리 비교용 신호정의, 0~9는 숫자와 동일값 *=10, #=11, A=12,B=13,C=14,D=15,null=16
signal set_no1 : integer range 0 to 20:=18; --set_no1신호선언, 초기값은18
signal set_no2 : integer range 0 to 20:=18; --set_no2신호선언, 초기값은18
signal set_no3 : integer range 0 to 20:=18; --set_no3신호선언, 초기값은18
signal set_no4 : integer range 0 to 20:=18; --set_no4신호선언, 초기값은18
signal set_no5 : integer range 0 to 20:=18; --set_no5신호선언, 초기값은18
signal set_no6 : integer range 0 to 20:=18; --set_no6신호선언, 초기값은18
signal save_no1 : integer range 0 to 17:=2; --save_no1신호선언, 초기값은 2
signal save_no2 : integer range 0 to 17:=5; --save_no2신호선언, 초기값은 5
signal save_no3 : integer range 0 to 17:=8; --save_no3신호선언, 초기값은 8
signal save_no4 : integer range 0 to 17:=0; --save_no4신호선언, 초기값은 0
signal save_no5 : integer range 0 to 17:=1; --save_no5신호선언, 초기값은 1
signal save_no6 : integer range 0 to 17:=3; --save_no6신호선언, 초기값은 3
--Door-lock password OK 신호
signal pass_ok : std_logic:='1'; --pass_ok,1이면 비밀번호 OK
-- 7-Segment관련 신호선언
signal fnd_clk : std_logic;    --7segment clock
signal fnd_cnt : integer range 3 downto 0; --fnd count에 사용
signal seg_1000 : std_logic_vector(6 downto 0); --segment 1000의 자리
signal seg_100 : std_logic_vector(6 downto 0);  --segment 100의 자리
signal seg_10 : std_logic_vector(6 downto 0); --segment 10의 자리
signal seg_1 : std_logic_vector(6 downto 0);  --segment 1의 자리


begin

npb <= not pb; --푸시버튼스위치는 눌렀을때 '0'이므로 not을 해준다.

process(clock_50m)  --감지신호가 clock_50m 인 process문
begin
 if rising_edge(clock_50m) then    --clock_50m이 상승에지 일때
   init_counter <= init_counter+1;  --init_counter 하나씩 증가함.
 end if;
end process;

--스위치를 눌렀을때 glitch가 발생하며 한번 눌렀을때 여러차례 신호가 감지된다.
--이를 보완하기 위해 버튼 스위치는 클럭으로 감지한다.
--본 예제에서는 40ms 클럭을 사용하고 두번 감지하여 같을때 눌린것으로 인식한다.
--참고로 버튼을 사람이 눌렀을때 대략 30ms이상이 된다.

sw_clk <= init_counter(20); -- ~40ms clock,
fnd_clk <= init_counter(16); -- ~2ms clock, 50MHz 클럭을 17번 분주한 clock

process(sw_clk,npb,pb_1st,pb_2nd,sw_toggle)  -- sw_clk,npb,pb_1st,PB_2nd,pb_toggle을 감지 신호로 하는 process문
begin
 if rising_edge(sw_clk)then -- sw_clk가 상승에지일때,~40ms 클럭으로 눌림 감지

	pb_2nd <= pb_1st; --pb_2nd에 pb_1st를 대입
   pb_1st <= npb;    --pb_1st에 npb를 대입
   
--	if(sw_status = sw_idle)then
--	  set_no1 <= 20; --set_no1에 20 대입
--	  set_no2 <= 20; --set_no2에 20 대입
--	  set_no3 <= 20; --set_no3에 20 대입
--	  set_no4 <= 20; --set_no4에 20 대입
--	end if;
   
	if((pb_2nd = X"0000") and (pb_1st /= pb_2nd))then --pb_2nd가 X"0000"이고 pb_1st와 pb_2nd가 다른값 일때
		sw_toggle <= '1';  -- sw_toggle에 1을 대입, 즉 버튼이 눌렸다면
  	end if;
  	
   if((sw_toggle = '1')and(pb_1st = pb_2nd))then --sw_toggle이 '1'이고 pb_1st와 pb_2nd가 같다면
		sw_toggle <= '0';  --sw_toggle에 '0'을 대입, 스위치를 계속 눌렀을때 연속 검출 방지
     case pb_1st is  --pb_1st의 값이
      when x"0_0_0_1" =>	-- switch_1, led_red(1)
        if (sw_status = sw_idle)then --sw_status가 sw_idle이면
          set_no1 <= 16; --set_no1에 16 대입
          set_no2 <= 17; --set_no2에 17 대입
          set_no3 <= 17; --set_no3에 17 대입
          set_no4 <= 20; --set_no4에 20 대입
          pass_ok <= '0';
  	     elsif(sw_status = sw_start)then --sw_status가 sw_start이면
  	       sw_status <= sw_s1; --sw_status에 sw_s1대입
          set_no1 <= 1; --set_no1에 1 대입, 처음눌린번호
          pass_ok <= '0';
  	     elsif(sw_status = sw_s1)then --sw_status가 sw_s1이면
  	       sw_status <= sw_s2; --sw_status에 sw_s2대입
          set_no2 <= 1; --set_no2에 1 대입, 두번째 눌린번호
          pass_ok <= '0';
  	     elsif(sw_status = sw_s2)then --sw_status가 sw_s2이면
  	       sw_status <= sw_s3; --sw_status에 sw_s3대입
          set_no3 <= 1; --set_no3에 1 대입, 세번째 눌린번호
          pass_ok <= '0';
  	     elsif(sw_status = sw_s3)then --sw_status가 sw_s3이면
  	       sw_status <= sw_s4; --sw_status에 sw_s4대입
          set_no4 <= 1; --set_no4에 1 대입, 네번째 눌린번호
          pass_ok <= '0';
  	     elsif(sw_status = sw_s4)then --sw_status가 sw_s4이면
          sw_status <= sw_s5; --sw_status에 sw_s4대입
          set_no5 <= 1; --set_no5에 1 대입, 다섯번째 눌린번호
          pass_ok <= '0';
		  elsif(sw_status = sw_s5)then --sw_status가 sw_s5이면
          sw_status <= sw_s6; --sw_status에 sw_s6대입
          set_no6 <= 1; --set_no6에 1 대입, 여섯번째 눌린번호
          pass_ok <= '0';
		  elsif(sw_status = sw_s6)then --sw_status가 sw_s6이면
          sw_status <= sw_idle; --sw_status에 sw_idle대입
          set_no1 <= 16; --set_no1에 16 대입
          set_no2 <= 17; --set_no2에 17 대입
          set_no3 <= 17; --set_no3에 17 대입
          set_no4 <= 20; --set_no4에 20 대입
          pass_ok <= '0';
		  elsif(sw_status = sw_save)then --sw_status가 sw_save이면
  	       sw_status <= sw_save1; --sw_status에 sw_save1대입
          set_no1 <= 1; --set_no1에 1 대입, 처음눌린번호
			 save_no1 <= 1; --save_no2에 1 대입, 비밀번호 첫 번째 번호
          pass_ok <= '0';
  	     elsif(sw_status = sw_save1)then --sw_status가 sw_save1이면
  	       sw_status <= sw_save2; --sw_status에 sw_save2대입
          set_no2 <= 1; --set_no2에 1 대입, 두번째 눌린번호
			 save_no2 <= 1;--save_no2에 1 대입, 비밀번호 두 번째 번호
          pass_ok <= '0';
  	     elsif(sw_status = sw_save2)then --sw_status가 sw_save2이면
  	       sw_status <= sw_save3; --sw_status에 sw_save3대입
          set_no3 <= 1; --set_no3에 1 대입, 세번째 눌린번호
			 save_no3 <= 1;--save_no3에 1 대입, 비밀번호 세 번째 번호
          pass_ok <= '0';
  	     elsif(sw_status = sw_save3)then --sw_status가 sw_save3이면
  	       sw_status <= sw_save4; --sw_status에 sw_save4대입
          set_no4 <= 1; --set_no4에 1 대입, 네번째 눌린번호
			 save_no4 <= 1;--save_no4에 1 대입, 비밀번호 네 번째 번호
          pass_ok <= '0';
  	     elsif(sw_status = sw_save4)then --sw_status가 sw_save4이면
          sw_status <= sw_save5; --sw_status에 sw_save5대입
          set_no5 <= 1; --set_no5에 1 대입, 다섯번째 눌린번호
			 save_no5 <= 1;--save_no5에 1 대입, 비밀번호 다섯 번째 번호
          pass_ok <= '0';
		  elsif(sw_status = sw_save5)then --sw_status가 sw_save5이면
          sw_status <= sw_save6; --sw_status에 sw_save6대입
          set_no6 <= 1; --set_no6에 1 대입, 여섯번째 눌린번호
			 save_no6 <= 1;--save_no6에 1 대입, 비밀번호 여섯 번째 번호
          pass_ok <= '0';
		  elsif(sw_status = sw_save6)then --sw_status가 sw_save6이면
          sw_status <= sw_idle; --sw_status에 sw_idle대입
          set_no1 <= 11; --set_no1에 16 대입
          set_no2 <= 12; --set_no2에 17 대입
          set_no3 <= 5; --set_no3에 17 대입
          set_no4 <= 5; --set_no4에 20 대입
          pass_ok <= '0';
        else
          null;
        end if;
      when x"0_0_0_2" =>	-- switch_2, led_red(3)
        if (sw_status = sw_idle)then --sw_status가 sw_idle이면
          set_no1 <= 16; --set_no1에 16 대입
          set_no2 <= 17; --set_no2에 17 대입
          set_no3 <= 17; --set_no3에 17 대입
          set_no4 <= 20; --set_no4에 20 대입
          pass_ok <= '0';
  	     elsif(sw_status = sw_start)then --sw_status가 sw_start이면
  	       sw_status <= sw_s1; --sw_status에 sw_s1대입
          set_no1 <= 2; --set_no1에 2 대입, 처음눌린번호
          pass_ok <= '0';
  	     elsif(sw_status = sw_s1)then --sw_status가 sw_s1이면
  	       sw_status <= sw_s2; --sw_status에 sw_s2대입
          set_no2 <= 2; --set_no2에 2 대입, 두번째 눌린번호
          pass_ok <= '0';
  	     elsif(sw_status = sw_s2)then --sw_status가 sw_s2이면
  	       sw_status <= sw_s3; --sw_status에 sw_s3대입
          set_no3 <= 2; --set_no3에 2 대입, 세번째 눌린번호
          pass_ok <= '0';
  	     elsif(sw_status = sw_s3)then --sw_status가 sw_s3이면
  	       sw_status <= sw_s4; --sw_status에 sw_s4대입
          set_no4 <= 2; --set_no4에 2 대입, 네번째 눌린번호
          pass_ok <= '0';
  	     elsif(sw_status = sw_s4)then --sw_status가 sw_s4이면
          sw_status <= sw_s5; --sw_status에 sw_s4대입
          set_no5 <= 2; --set_no5에 2 대입, 다섯번째 눌린번호
          pass_ok <= '0';
		  elsif(sw_status = sw_s5)then --sw_status가 sw_s5이면
          sw_status <= sw_s6; --sw_status에 sw_s6대입
          set_no6 <= 2; --set_no6에 2 대입, 여섯번째 눌린번호
          pass_ok <= '0';
		  elsif(sw_status = sw_s6)then --sw_status가 sw_s6이면
          sw_status <= sw_idle; --sw_status에 sw_idle대입
          set_no1 <= 16; --set_no1에 16 대입
          set_no2 <= 17; --set_no2에 17 대입
          set_no3 <= 17; --set_no3에 17 대입
          set_no4 <= 20; --set_no4에 20 대입
          pass_ok <= '0';
		  elsif(sw_status = sw_save)then --sw_status가 sw_save이면
  	       sw_status <= sw_save1; --sw_status에 sw_save1대입
          set_no1 <= 2; --set_no1에 2 대입, 처음눌린번호
			 save_no1 <= 2; --save_no2에 2 대입, 비밀번호 첫 번째 번호
          pass_ok <= '0';
  	     elsif(sw_status = sw_save1)then --sw_status가 sw_save1이면
  	       sw_status <= sw_save2; --sw_status에 sw_save2대입
          set_no2 <= 2; --set_no2에 2 대입, 두번째 눌린번호
			 save_no2 <= 2;--save_no2에 2 대입, 비밀번호 두 번째 번호
          pass_ok <= '0';
  	     elsif(sw_status = sw_save2)then --sw_status가 sw_save2이면
  	       sw_status <= sw_save3; --sw_status에 sw_save3대입
          set_no3 <= 2; --set_no3에 2 대입, 세번째 눌린번호
			 save_no3 <= 2;--save_no3에 2 대입, 비밀번호 세 번째 번호
          pass_ok <= '0';
  	     elsif(sw_status = sw_save3)then --sw_status가 sw_save3이면
  	       sw_status <= sw_save4; --sw_status에 sw_save4대입
          set_no4 <= 2; --set_no4에 2 대입, 네번째 눌린번호
			 save_no4 <= 2;--save_no4에 2 대입, 비밀번호 네 번째 번호
          pass_ok <= '0';
  	     elsif(sw_status = sw_save4)then --sw_status가 sw_save4이면
          sw_status <= sw_save5; --sw_status에 sw_save5대입
          set_no5 <= 2; --set_no5에 2 대입, 다섯번째 눌린번호
			 save_no5 <= 2;--save_no5에 2 대입, 비밀번호 다섯 번째 번호
          pass_ok <= '0';
		  elsif(sw_status = sw_save5)then --sw_status가 sw_save5이면
          sw_status <= sw_save6; --sw_status에 sw_save6대입
          set_no6 <= 2; --set_no6에 2 대입, 여섯번째 눌린번호
			 save_no6 <= 2;--save_no6에 2 대입, 비밀번호 여섯 번째 번호
          pass_ok <= '0';
		 elsif(sw_status = sw_save6)then --sw_status가 sw_save6이면
          sw_status <= sw_idle; --sw_status에 sw_idle대입
          set_no1 <= 11; --set_no1에 16 대입
          set_no2 <= 12; --set_no2에 17 대입
          set_no3 <= 5; --set_no3에 17 대입
          set_no4 <= 5; --set_no4에 20 대입
          pass_ok <= '0';
        else
          null;
        end if;
      when x"0_0_0_4" =>	-- switch_3, led_red(2)
        if (sw_status = sw_idle)then --sw_status가 sw_idle이면
          set_no1 <= 16; --set_no1에 16 대입
          set_no2 <= 17; --set_no2에 17 대입
          set_no3 <= 17; --set_no3에 17 대입
          set_no4 <= 20; --set_no4에 20 대입
          pass_ok <= '0';
  	     elsif(sw_status = sw_start)then --sw_status가 sw_start이면
  	       sw_status <= sw_s1; --sw_status에 sw_s1대입
          set_no1 <= 3; --set_no1에 3 대입, 처음눌린번호
          pass_ok <= '0';
  	     elsif(sw_status = sw_s1)then --sw_status가 sw_s1이면
  	       sw_status <= sw_s2; --sw_status에 sw_s2대입
          set_no2 <= 3; --set_no2에 3 대입, 두번째 눌린번호
          pass_ok <= '0';
  	     elsif(sw_status = sw_s2)then --sw_status가 sw_s2이면
  	       sw_status <= sw_s3; --sw_status에 sw_s3대입
          set_no3 <= 3; --set_no3에 3 대입, 세번째 눌린번호
          pass_ok <= '0';
  	     elsif(sw_status = sw_s3)then --sw_status가 sw_s3이면
  	       sw_status <= sw_s4; --sw_status에 sw_s4대입
          set_no4 <= 3; --set_no4에 3 대입, 네번째 눌린번호
          pass_ok <= '0';
  	     elsif(sw_status = sw_s4)then --sw_status가 sw_s4이면
          sw_status <= sw_s5; --sw_status에 sw_s4대입
          set_no5 <= 3; --set_no5에 3 대입, 다섯번째 눌린번호
          pass_ok <= '0';
		  elsif(sw_status = sw_s5)then --sw_status가 sw_s5이면
          sw_status <= sw_s6; --sw_status에 sw_s6대입
          set_no6 <= 3; --set_no6에 3 대입, 여섯번째 눌린번호
          pass_ok <= '0';
		  elsif(sw_status = sw_s6)then --sw_status가 sw_s6이면
          sw_status <= sw_idle; --sw_status에 sw_idle대입
          set_no1 <= 16; --set_no1에 16 대입
          set_no2 <= 17; --set_no2에 17 대입
          set_no3 <= 17; --set_no3에 17 대입
          set_no4 <= 20; --set_no4에 20 대입
          pass_ok <= '0';
		  elsif(sw_status = sw_save)then --sw_status가 sw_save이면
  	       sw_status <= sw_save1; --sw_status에 sw_save1대입
          set_no1 <= 3; --set_no1에 3 대입, 처음눌린번호
			 save_no1 <= 3; --save_no2에 3 대입, 비밀번호 첫 번째 번호
          pass_ok <= '0';
  	     elsif(sw_status = sw_save1)then --sw_status가 sw_save1이면
  	       sw_status <= sw_save2; --sw_status에 sw_save2대입
          set_no2 <= 3; --set_no2에 3 대입, 두번째 눌린번호
			 save_no2 <= 3;--save_no2에 3 대입, 비밀번호 두 번째 번호
          pass_ok <= '0';
  	     elsif(sw_status = sw_save2)then --sw_status가 sw_save2이면
  	       sw_status <= sw_save3; --sw_status에 sw_save3대입
          set_no3 <= 3; --set_no3에 3 대입, 세번째 눌린번호
			 save_no3 <= 3;--save_no3에 3 대입, 비밀번호 세 번째 번호
          pass_ok <= '0';

  fnd_cnt <= fnd_cnt+1;       --fnd_cnt를 1씩 증가 시킨다.
  case fnd_cnt is             --fnd_cnt 값을 확인하여
   when 3 =>                  --3일때 
     fnd_d <= '0' & seg_1000; --fnd_d에 seg_1000을 대입 한다.
   when 2 =>                  --2일때

   when 1 =>                  --3일때

   when 0 =>                  --0일때

   when others =>             --기타 값일때 변화 없음
     null;
  end case;
 
  if(fnd_cnt = 3)then  --fnd_cnt 값이 3일때 
    fnd_s(3) <= '0';   --fnd_s(3)에 '0' 대입
  else             --fnd_cnt 값이 3이 아닐때
