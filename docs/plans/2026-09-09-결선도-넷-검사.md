## 배경 — 왜 이 결함이 검사를 통과했나

2026-09-09 사용자가 복합 회로 3단계 그림의 ⑯ 점퍼(`우 a10 → 우 b10`)를 짚었다. 넷을 추적해 실험01·02 결선도 13장을 전부 대조한 결과 결함이 둘 나왔다.

**1. 복합 회로 74HC32 pin 1 이 뜬다.** `우 a10` 과 `우 b10` 은 같은 5홀 스트립이라 이 점퍼는 전기적으로 아무 일도 하지 않는다. 그런데 `a10` 으로 입력 A 를 가져오는 선이 없다 — ⑮ 갈래는 여백 `x=577` 로 내려와 `a13`(pin 4 행)에만 꽂힌다. 그림대로 꽂으면 pin 1 이 뜨고 `X = A + Z` 가 성립하지 않아, 넷리스트가 약속한 "LED_X 와 LED_Z 는 언제나 같이 켜진다"는 자체 검산이 깨진다.

이력이 중요하다. 최초 판(커밋 8718349)에서는 ⑮ 가 오른쪽 a열을 `a5 → a13` 으로 타고 내려가며 **a10 위를 지났고**, ⑯ 은 그 세로선에서 옆 홀로 빼는 T 접속처럼 보였다. 2026-09-08 에 R3(끝점 위 통과)를 통과시키려고 ⑮ 를 여백으로 옮겼는데, 그 재배치가 a10 으로 오던 유일한 (시각적) 공급을 끊었고 ⑯ 은 손대지 않아 무의미한 토막으로 남았다. **기하 검사를 만족시키려는 재배치가 넷을 끊었고, 검사는 오히려 "통과"로 바뀌었다.**

**2. 실험02 Rn 그림이 +5V 를 GND 로 단락시킨다.** `_build_e2_dc` 가 1kΩ 을 `a11 → a14` 로 3행 길이로 그리는데 14행은 74HC04 **pin 5 의 행**이다. 그 스트립에 저항 아랫다리(`a14`), "R1 반대쪽 → +5V" 점퍼(`b14`), pin 5 미사용 입력 접지선(`d14 → GND`)이 함께 꽂힌다. `+5V 레일 → b14 → 스트립 → d14 → GND 레일` 이 직결된다. 전원장치가 전류 제한에 걸리거나 퓨즈가 나가고, pin 5 도 넷리스트(GND)와 달리 VCC 에 물린다. Rp 는 양쪽이 다 GND 라 단락은 없지만 저항 다리가 칩 행에 앉는 구조는 똑같다. 이쪽은 부품 길이를 3행으로 고른 첫 판부터 있었고 아무도 건드리지 않아 살아남았다.

**둘 다 시험 128건과 `verify()` R1~R4 를 전부 통과한 상태로 커밋돼 있었다.** R1~R4 는 끝점이 홀인가, 한 홀에 리드 하나인가, 끝점 위를 지나는가, 축이 포개지는가만 본다. "같은 스트립 안에서 끝나는 점퍼", "한 넷에 VCC 와 GND", "공급 없는 핀"은 아무도 보지 않는다. 손그림 검토(`review-hand-diagram`)는 `wiring()` 을 정답으로 읽으므로 틀린 참조를 틀린 채로 통과시킨다 — 하류에서 잡힐 기회도 없었다.

> 2026-09-07 의 교훈이 그대로 반복됐다. 그때는 "규칙을 문서에 더 쓰는 대신 검사에 넣었다 — 문서를 읽어선 안 나온다"였다. 이번에는 **넣은 검사가 기하뿐이었다.** 그래서 R5 를 넣는다.

```python
브레드보드는 홀이 아니라 **스트립** 단위로 이어진다. 단자대는 한 행의 a~e

"""
import importlib.util
import os
import tempfile
import unittest

REPO_ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
SCRIPT = os.path.join(REPO_ROOT, '.claude', 'skills', 'breadboard-diagram',
                      'scripts', 'render_breadboard.py')


def _load():
    spec = importlib.util.spec_from_file_location('render_breadboard', SCRIPT)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod


def _render(mod, fn):
    """빌더를 임시 파일로 렌더하고 렌더러 객체를 돌려준다."""
    captured = {}
    orig = mod.BreadboardRenderer.export

    def spy(self_, *a, **k):
        captured['bb'] = self_
        return orig(self_, *a, **k)

    mod.BreadboardRenderer.export = spy
    try:
        with tempfile.TemporaryDirectory() as d:
            fn(os.path.join(d, 'x.png'))
    finally:
        mod.BreadboardRenderer.export = orig
    return captured['bb']


def _net_of(bb, label):
    for net in bb.nets():
        if label in net['labels']:
            return net
    raise AssertionError(f'{label} 이 어느 넷에도 없다')


class NetModel(unittest.TestCase):
    """스트립을 옳게 묶는가. 합성 보드에 몇 줄만 그리고 `nets()` 만 부른다."""

    @classmethod
    def setUpClass(cls):
        cls.mod = _load()

    def board(self):
        bb = self.mod.BreadboardRenderer()
        bb.enable_steps()
        return bb

    def test_같은_행_a부터_e_는_한_스트립이다(self):
        bb = self.board()
        c = bb.cols['left']
        bb.draw_chip('74HC08', 10)
        bb.step_wire([(c['d'], bb.r_y(10)), (c['d'], bb.r_y(13))], 'sig', '시험')
        self.assertIn('74HC08#10 pin 4', _net_of(bb, '74HC08#10 pin 1')['labels'])

    def test_칩_건너편_f부터_j_는_다른_스트립이다(self):
        bb = self.board()
        bb.draw_chip('74HC08', 10)
        self.assertNotIn('74HC08#10 pin 14', _net_of(bb, '74HC08#10 pin 1')['labels'])

    def test_레일쌍은_브리지_전에는_다른_넷이다(self):
        bb = self.board()
        c = bb.cols['left']
        self.assertNotIn('좌 중앙GND레일', _net_of(bb, '좌 GND레일')['labels'])
        bb.bridge_rails(c['gnd'], c['mid_gnd'], 29, kind='gnd')
        self.assertIn('좌 중앙GND레일', _net_of(bb, '좌 GND레일')['labels'])

    def test_입력_점퍼의_레일_끝은_가상_노드다(self):
        """입력은 지금 GND 레일에 꽂혀 있지만(=0 자리) 1 을 줄 때 그 끝만
        VCC 로 옮긴다. 레일 연결로 세면 입력 A·B·C 가 GND 와 한 넷이 되어
        'pin 1 이 A 에 물렸는가' 를 물을 수 없다."""
        bb = self.board()
        c = bb.cols['left']
        bb.draw_chip('74HC08', 10)
        self.mod._input_jumper(bb, c, 10, 'A', '74HC08 pin 1')
        net = _net_of(bb, '입력 A')
        self.assertIn('74HC08#10 pin 1', net['labels'])
        self.assertNotIn('좌 GND레일', net['labels'])


if __name__ == '__main__':
    unittest.main()
```

`def verify(self):` 바로 위. 최종 형태:

```python
    def input_node(self, name, x, row):
        """**방금 그린 배선**을 입력 점퍼로 적어 둔다 (`_input_jumper` 가 부른다).

        입력 점퍼는 지금 GND 레일에 꽂혀 있다 — 0 을 준 자리를 그린 것이고,
        1 을 줄 때는 레일 쪽 끝만 VCC 레일로 옮긴다. 넷 검사가 이걸 레일
        연결로 세면 입력 A·B·C 가 GND 와 한 넷이 되어 "pin 1 이 입력 A 에
        물렸는가" 를 물을 수 없다. 그래서 그 끝을 가상 입력 노드로 바꿔 센다.
        """
        self.inputs.append({'name': name, 'at': self.sym(x, row),
                            'seg': len(self._segments) - 1})

    _AE = ('a', 'b', 'c', 'd', 'e')
    _FJ = ('f', 'g', 'h', 'i', 'j')

    def _strip(self, sym):
        """기호 좌표 -> 스트립 키. **홀이 아니라 스트립이 넷의 단위다.**

        단자대는 한 행의 a~e 가 한 스트립, f~j 가 또 한 스트립이다 (가운데
        골이 갈라 놓는다). 전원 레일은 열 하나가 통째로 한 스트립이라 행이
        없다 — 레일쌍 넷이 서로 별개 스트립인 것은 x 가 다르기 때문이다.
        """
        if sym is None:
            return None
        block, col, row = sym
        if col in self._AE:
            return (block, 'ae', row)
        if col in self._FJ:
            return (block, 'fj', row)
        return ('rail', self.cols[block][col])

    def _pin_label(self, ch, p):
        """칩 핀의 넷 라벨. 한 그림에 같은 칩이 둘이면(`e2_tpd`) 행으로 갈린다."""
        return f"{ch['label']}#{ch['pin1_row']} pin {p}"

    def nets(self):
        """그린 것을 **전기적 넷**으로 묶어 낸다.

        -> `[{'labels': {'74HC32#10 pin 1', '입력 A', ...}, 'rails': {'vcc'}}, ...]`

        `_leads` 는 홀 단위라 이 층위를 못 본다 — 같은 스트립의 다른 홀에 꽂은
        두 점퍼는 홀이 다르지만 **전기적으로 한 점**이다. 2026-09-09 의 결함
        둘이 정확히 그 틈으로 지나갔다.

        `desc` 문자열은 읽지 않는다. `chips`·`leds`·`resistors`·`_segments`·
        `inputs` 라는 **구조**만 읽는다 — 말이 바뀔 때 조용히 깨지지 않도록.
        """
        uf = {}

        def find(k):
            uf.setdefault(k, k)
            while uf[k] != k:
                uf[k] = uf[uf[k]]
                k = uf[k]
            return k

        def union(a, b):
            ra, rb = find(a), find(b)
            if ra != rb:
                uf[rb] = ra

        members = {}

        def put(strip, label):
            if strip is not None:
                members.setdefault(strip, set()).add(label)
                find(strip)

        for ch in self.chips:
            for p, sym in ch['pins'].items():
                put(self._strip(sym), self._pin_label(ch, p))
        for led in self.leds:
            put(self._strip(led['anode']), f"LED {led['tag']} 애노드")
            put(self._strip(led['cathode']), f"LED {led['tag']} 캐소드")
        for i, r in enumerate(self.resistors, 1):
            put(self._strip(r['top']), f"저항{i}({r['label']}) 윗다리")
            put(self._strip(r['bottom']), f"저항{i}({r['label']}) 아랫다리")
        for x, name in self._addr_map.items():
            if x in self._rail_xs():
                put(('rail', x), name)

        # 입력 점퍼는 레일 쪽 끝을 가상 노드로 바꾼다 (`input_node()` 참조)
        sub = {}
        for inp in self.inputs:
            sub[inp['seg']] = ('input', inp['name'])
            put(('input', inp['name']), f"입력 {inp['name']}")

        for i, (kind, pts, desc) in enumerate(self._segments):
            ends = []
            for p in (pts[0], pts[-1]):
                if p[1] < 240 and any(abs(p[0] - jx) < 30 for jx in JACKS.values()):
                    color = min(JACKS, key=lambda k: abs(JACKS[k] - p[0]))
                    ends.append(('jack', color))
                    put(('jack', color), f'{JACK_KR[color]} 잭')
                else:
                    ends.append(self._strip(self.sym(p[0], self._row_of(p[1]))))
            if i in sub:
                ends = [sub[i] if e is not None and e[0] == 'rail' else e for e in ends]
            if ends[0] is None or ends[1] is None:
                continue   # 홀이 아닌 끝점은 R1 이 따로 잡는다
            union(ends[0], ends[1])

        groups = {}
        for strip in list(uf):
            groups.setdefault(find(strip), []).append(strip)
        out = []
        for strips in groups.values():
            labels = set()
            rails = set()
            for st in strips:
                labels |= members.get(st, set())
                if st[0] == 'rail':
                    rails.add(self._rail_kind.get(st[1]))
            out.append({'labels': labels, 'rails': rails - {None}})
        return out

    def verify(self):
```

```python
                if (ch['block'], ch['pin1_row'], p) in open_pins:
                    continue
                if self._strip(sym) in probe_strips:
                    continue
                lab = self._pin_label(ch, p)
                for net in nets:
                    if lab in net['labels']:
                        if len(net['labels']) == 1:
                            r5.append(f'{lab} 이 아무 데도 안 물렸다 — 뜬 핀')
                        break

        if r5:
            raise AssertionError('넷이 틀렸다:\n  ' + '\n  '.join(r5))

        # 레일 연결성. 같은 전원을 나르는 레일들이 브리지로 이어져 있지
```
