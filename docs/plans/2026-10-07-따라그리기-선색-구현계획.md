# 따라그리기 층 쪽 선마다 다른 색 구현계획

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 따라그리기 PDF 의 층 쪽에서 이번에 그릴 신호선을 선마다 다른 색으로 그리고(선·끝점·번호 원·표 번호 칸 같은 색), 실험05 1-2 에 LED 순서 안내를 더하고 겹치는 층을 쪼갠다.

**Architecture:** 엔진 `step_wire` 가 층 렌더(`layer` 있음)에서 shown 인 `sig`·`mid` 선에 `LAYER_PALETTE` 색을 등장 순서대로 주고 `bb.layer_colors = {번호: RGBA}` 에 기록한다. `trace_guide` 는 층 보드의 `layer_colors` 를 읽어 표 번호 칸을 같은 색으로 칠하고, 10색을 넘는 층은 `LayerError` 로 멈춘다. 완성본·단계 그림(`layer is None`)은 픽셀 하나 안 바뀐다.

**Tech Stack:** Python 3.14, Pillow, `unittest`(pytest·xdist 로 실행, 한 파일 볼 때 `-n0`), pypdf, 헤드리스 Chrome(`html2pdf`).

**Spec:** `docs/plans/2026-10-07-따라그리기-선색-설계.md`

## Global Constraints

- 색칠 대상: `self.layer is not None` · `self.numbered` · view `'shown'` · `kind in ('sig', 'mid')` · `color` 인자 None — 넷 다 참일 때만.
- `+5V` 빨강 `WIRE['vcc']`·GND 파랑 `WIRE['gnd']`·입력 보라 `INPUT`·회색 `KEEP` 은 바꾸지 않는다.
- `LAYER_PALETTE` 는 10색이고 `WIRE`·`INPUT`·`KEEP`·`PROBE`·`REMOVE` 어느 색과도 같지 않다.

- 엔진 코드 변경은 주차 결선과 따로 커밋한다(`breadboard-diagram` SKILL §3).
- 실험05 에서 다시 굽는 그림은 C1(1-2) 다섯 장뿐이다. C2 PNG·JSON 은 커밋본 그대로 둔다.

```python
선·끝점·번호 원이 같은 색이라 같은 색 점끼리 이으면 된다. 완성본(layer 없음)은 그대로다.

import importlib.util
import os
import unittest

REPO_ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
SCRIPT = os.path.join(REPO_ROOT, '.claude', 'skills', 'breadboard-diagram',
                      'scripts', 'render_breadboard.py')


def _load():
    spec = importlib.util.spec_from_file_location('render_breadboard', SCRIPT)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod


def _pixel(bb, x, y):
    """템플릿 좌표 (x, y) 의 overlay 픽셀."""
    k = bb.overlay.size[0] / bb.base_im.size[0]
    return bb.overlay.getpixel((round(x * k), round(y * k)))


def _mid(bb, seg_index):
    _kind, pts, _d = bb._segments[seg_index]
    (x0, y0), (x1, y1) = pts[0], pts[1]
    return (x0 + x1) / 2, (y0 + y1) / 2


class LayerColor(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.mod = _load()

    def board(self, layer=None, n_sig=3):
        """잭 두 가닥(1·2번) + 세로 신호선 n_sig 가닥(3번부터)."""
        bb = self.mod.BreadboardRenderer(board='v1', layer=layer)
        bb.enable_steps()
        bb.set_header('층 색', '테스트')
        bb.connect_jacks()
        c = bb.cols['left']
        for i in range(n_sig):
            bb.step_wire([(c['a'], bb.r_y(10 + 2 * i)), (c['a'], bb.r_y(11 + 2 * i))], 'sig', f'신호 {i}')
        return bb

    def test_팔레트는_열_색이고_넷_색과_겹치지_않는다(self):
        p = self.mod.LAYER_PALETTE
        self.assertEqual(len(p), 10)
        self.assertEqual(len(set(p)), 10)
        taken = set(self.mod.WIRE.values()) | {self.mod.INPUT, self.mod.KEEP, self.mod.PROBE, self.mod.REMOVE}
        self.assertFalse(set(p) & taken)

    def test_층_신호선은_선마다_다른_팔레트_색이다(self):
        bb = self.board((3, 5))
        self.assertEqual(bb.layer_colors, {3: self.mod.LAYER_PALETTE[0], 4: self.mod.LAYER_PALETTE[1],
                                           5: self.mod.LAYER_PALETTE[2]})
        for i, n in enumerate((3, 4, 5)):
            self.assertEqual(_pixel(bb, *_mid(bb, n - 1)), self.mod.LAYER_PALETTE[i])

    def test_전원선은_층에서도_빨강_파랑이다(self):
        bb = self.board((1, 5))
        self.assertEqual(_pixel(bb, *_mid(bb, 0)), self.mod.WIRE['vcc'])
        self.assertEqual(_pixel(bb, *_mid(bb, 1)), self.mod.WIRE['gnd'])
        self.assertEqual(sorted(bb.layer_colors), [3, 4, 5])

    def test_끝점_점이_커진다(self):
        bb = self.board((3, 3), n_sig=1)
        _kind, pts, _d = bb._segments[2]
        x, y = pts[0]
        self.assertEqual(_pixel(bb, x + 4, y), self.mod.LAYER_PALETTE[0])   # 반지름 5 안, 선 폭(3) 밖

    def test_층_없으면_팔레트를_안_쓴다(self):
        bb = self.board(None)
        self.assertEqual(bb.layer_colors, {})
        self.assertEqual(_pixel(bb, *_mid(bb, 2)), self.mod.WIRE['sig'])
        _kind, pts, _d = bb._segments[2]
        x, y = pts[0]
        self.assertEqual(_pixel(bb, x + 4, y)[3], 0)   # 완성본 끝점 반지름 3 그대로

    def test_앞뒤_층_신호선은_팔레트를_안_받는다(self):
        bb = self.board((4, 4))
        self.assertEqual(bb.layer_colors, {4: self.mod.LAYER_PALETTE[0]})
        self.assertEqual(_pixel(bb, *_mid(bb, 2)), self.mod.KEEP)
        self.assertEqual(_pixel(bb, *_mid(bb, 4))[3], 0)

    def test_입력_점퍼는_층에서도_보라다(self):
        bb = self.mod.BreadboardRenderer(board='v1', layer=(3, 20))   # 입력 점퍼가 번호를 몇 개 쓰든 출력선까지 덮는다
        bb.enable_steps()
        bb.connect_jacks()
        bb.draw_chip('74HC08', 10)
        self.mod._input_jumper(bb, bb.cols['left'], 10, 'A', '74HC08 pin 1')
        c = bb.cols['left']
        bb.step_wire([(c['a'], bb.r_y(30)), (c['a'], bb.r_y(31))], 'sig', '출력')
        i = next(i for i, s in enumerate(bb._segments) if '입력 A' in s[2])
        self.assertEqual(_pixel(bb, *_mid(bb, i)), self.mod.INPUT)
        self.assertEqual(list(bb.layer_colors.values()), [self.mod.LAYER_PALETTE[0]])

    def test_층_색은_도장에_없다(self):
        a, b = self.board((3, 5)), self.board((3, 5))
        b.layer_colors.clear()
        self.assertEqual(a.content(), b.content())
        self.assertEqual(self.board(None).sidecar()['steps'], a.sidecar()['steps'])

    unittest.main()
```

```python
            part = fn(str(work_dir / name), None, layer=(lo, hi))
            colors = dict(part.layer_colors)
            if len(colors) > palette:
                raise LayerError(f'{filename}: 층 "{label}" 의 색 신호선이 {len(colors)}가닥인데 팔레트는 '
                                 f'{palette}색이다 — 같은 색이 두 번 나온다. 그 층을 쪼갠다')
            rows = [s for s in steps if lo <= s['n'] <= hi]
            pages.append(Page(name, f'{final.title} — {i}/{total} · {label} ({lo}~{hi}번)',
                              rows, [], legend, colors))
```
