```python
# tmp/실험03_균형/run.py — 시험 빌더를 뽑고 잰다. 저장소 테스트 밖이다.
import importlib.util
import os
import sys

HERE = os.path.dirname(os.path.abspath(__file__))
ROOT = os.path.dirname(os.path.dirname(HERE))
ENGINE = os.path.join(ROOT, '.claude', 'skills', 'breadboard-diagram', 'scripts', 'render_breadboard.py')


def load_engine():
    spec = importlib.util.spec_from_file_location('render_breadboard', ENGINE)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod


def measure(label, bb):
    # 한도 표시는 넣지 않는다 — 엔진이 R7 로 이미 막으므로 뽑힌 그림은 전부 ≤ 8 이다
    print(f'== {label}: 가닥 {len(bb.steps)}')
    for s in bb.side_leads():
        blk = '좌' if s['block'] == 'left' else '우'
        side = '왼쪽' if s['side'] == 'a' else '오른쪽'
        print(f"   {s['chip']:7s} {blk} {s['pin1_row']:2d}행 {side} {s['count']:2d}  {' '.join(s['addrs'])}")


def main():
    mod = load_engine()
    board = mod.load_board_file(os.path.join(HERE, '배치.py'), 'board_x3')
    made = mod.build_all(board, HERE)
    print('뽑음:', *made.keys(), sep='\n  ')
    for name, fn in board.FINALS:
        measure(f'시험 {name}', fn(os.path.join(HERE, '_measure_' + name)))
    if '--compare' in sys.argv:
        b3 = mod.load_board(3)
        for label, fn in (('승인본 전가산기', b3.build_e3_fa), ('승인본 전감산기', b3.build_e3_fs)):
            measure(label, fn(os.path.join(HERE, '_measure_승인본_' + fn.__name__ + '.png')))


if __name__ == '__main__':
    main()
```

```python
# 시험 — 실험03 전가산기·전감산기를 게이트 1·4(hex 1·6)로 다시 짠 것. tmp/ 라 커밋되지 않는다.
# 설계: docs/plans/2026-09-20-실험03-균형결선-격리시험-설계.md
# 승인본(보고서/실험03/결선도/배치.py)과의 차이는 게이트 배정과 그에 따른 배선 경로뿐이다.
# 슬롯·LED 자리·프레임 순서는 같다. density_policy 는 기본(strict) — 그래야 R7 이 재는 시험이다.
import os
import tempfile

from render_breadboard import BreadboardRenderer, _input_jumper, removal_board

X3_SLOTS = {'S1': ('left', 7), 'S2': ('left', 18), 'S3': ('right', 10), 'S4': ('right', 21)}
_S1, _S2, _S3, _S4 = (X3_SLOTS[k][1] for k in ('S1', 'S2', 'S3', 'S4'))
_LED1 = 25      # 좌 a25 사슬 — S / D
_LED2 = 29      # 우 c29 사슬 — Co / Bo
_LANE = 212     # 좌 여백 세로 차선 — LED1 급전
_LANE_A = 440   # 좌 j~중앙 레일 여백 — A' 건너감
_LANE_X = 452   # 같은 여백 — (A⊕B)' 건너감
_INPUT_NOTE = ('입력은 GND(=0) 에 꽂힌 자리로 그렸다. 1 을 줄 때는 그 점퍼의 레일 쪽 끝만 '
               '빨간 +5V 레일로 옮겨 꽂는다. 띄워두지 않는다.')
_LED_NOTE = ('LED 는 두 개뿐이다(교안 p8). LED1 = 합/차(S·D), LED2 = 올림/내림(Co·Bo). '
             '두 회로에서 LED·저항·급전선 자리가 같다.')
# 게이트 배정 — 쿼드는 1·4(핀 1,2→3 / 12,13→11), hex 는 1·6(1→2 / 13→12)
_QUAD14_UNUSED = ([4, 5, 9, 10], [6, 8])          # 게이트 2·3 미사용
_QUAD1_UNUSED = ([4, 5, 9, 10, 12, 13], [6, 8, 11])
_HEX16_UNUSED = ([3, 5, 9, 11], [4, 6, 8, 10])    # 인버터 2·3·4·5 미사용


def _x3_frame(bb):
    """잭 + 레일 브리지 넷. 승인본 셋에 좌 GND↔좌 중앙 GND 가 **하나 더** 있다 —
    Ci/Bi 입력이 f쪽(pin 13)이라 좌 중앙 GND 468 에 꽂히기 때문이다(`_input_jumper(side='f')`).
    이 브리지가 없으면 R5 연결성 검사가 '레일이 끊겨 있다' 로 막아야 정상이다 — 시험 항목."""
    c, r = bb.cols['left'], bb.cols['right']
    bb.connect_jacks()
    bb.bridge_rails(c['gnd'], r['mid_gnd'], 29, kind='gnd')
    bb.bridge_rails(c['gnd'], c['mid_gnd'], 31, kind='gnd')
    bb.bridge_rails(c['vcc'], c['mid_vcc'], 30, kind='vcc')
    bb.bridge_rails(c['mid_vcc'], r['vcc'], 34, kind='vcc')


def _x3_draw(bb, circuit, stage):
    """`circuit` 은 fa·fs, `stage` 는 1 또는 None(완성). 승인본 `_e3_draw` 와 같은 헬퍼 순서다 —
    순서: S1 칩 → 프레임 → S1·S3 전원 → 입력 A·B(a쪽)·Ci/Bi(f쪽) → XOR 사슬 → LED1 → 우측 AND
    → (fs) NOT → OR → LED2 → S4 전원 → S2 전원."""
    if stage == 1 and circuit != 'fa':
        raise ValueError(f'{circuit} 에는 1단계가 없다')
    c, r = bb.cols['left'], bb.cols['right']
    L = bb.r_y
    sub = circuit == 'fs'
    cin = 'Bi' if sub else 'Ci'
    sum_tag = 'D' if sub else 'S'
    carry_tag = 'Bo' if sub else 'Co'
    done = stage != 1
    bb.partial = not done

    # --- 좌 S1: 74HC86. XOR1 = 게이트1(a쪽), XOR2 = 게이트4(f쪽) ---
    bb.draw_chip('74HC86', _S1)
    _x3_frame(bb)
    bb.connect_power(_S1, gnd_col='b')
    bb.connect_power(_S3, block='right', gnd_col='a')
    _input_jumper(bb, c, 7, 'A', '74HC86 pin 1')
    _input_jumper(bb, c, 8, 'B', '74HC86 pin 2')
    # Ci/Bi → pin 13 (f쪽 8행). j 열에 꽂는다 — 레일 9행(8행은 빈 행)에서 오는 사선이 i8 옆을
    # 지나므로 갈래는 i8 에서 위로 뺀다(밑동 규칙)
    _input_jumper(bb, c, 8, cin, '74HC86 pin 13 (XOR2)', side='f', col='j')
    # A⊕B(pin 3, a쪽 9행) → XOR2 pin 12 (f쪽 9행): 칩 몸통 위를 안 지나고 칩 아래 14행을 돈다
    bb.step_wire([(c['c'], L(9)), (c['c'], L(14)), (c['h'], L(14)), (c['h'], L(9))], 'mid',
                 'XOR1 출력 A⊕B -> XOR2 입력 (pin 3 -> pin 12, 칩 아래로)', width=4,
                 marker_at=(c['f'], L(14) + 12))
    # S = pin 11 (f쪽 10행) → LED1(좌 a25): i10 → 15행 → 212 차선 → a25
    bb.step_wire([(c['i'], L(10)), (c['i'], L(15)), (_LANE, L(15)), (_LANE, L(_LED1)), (c['a'], L(_LED1))],
                 'sig', f'{sum_tag} = 74HC86 pin 11 -> LED1', marker_at=(_LANE + 13, L(20)))
    sum_desc = {'fa': 'S = A⊕B⊕Ci — 진리표 T3 의 S 열', 'fs': 'D = A⊕B⊕Bi — 진리표 T7 의 D 열'}[circuit]
    bb.led_chain(c['a'], _LED1, sum_tag, sum_desc, color_rgb=(210, 140, 0))

    if not done:
        bb.led_chain(r['c'], _LED2, carry_tag, '아직 급전하지 않는다 — 2단계에서 잇는다', block='right')
        bb.mark_unused_inputs(_QUAD14_UNUSED[0], _S1, chip='74HC86')
        bb.mark_unused_outputs(_QUAD14_UNUSED[1], _S1, chip='74HC86')
        bb.note('1단계: 74HC86 게이트 1·4 만 검사한다. LED1 = S = A⊕B⊕Ci 를 여덟 조합 전부 본다.')
        bb.note(_LED_NOTE)
        bb.note(_INPUT_NOTE)
        return

    # --- 우 S3: 74HC08. AND1 = 게이트1(a쪽), AND2 = 게이트4(f쪽) ---
    bb.draw_chip('74HC08', _S3, block='right')
    if not sub:
        bb.step_wire([(c['d'], L(7)), (c['d'], L(4)), (r['a'], L(4)), (r['a'], L(10))], 'sig',
                     '입력 A 갈래 -> 74HC08 pin 1', width=2, marker_at=(r['a'] - 14, L(7)))
    bb.step_wire([(c['c'], L(8)), (c['c'], L(5)), (r['b'], L(5)), (r['b'], L(11))], 'sig',
                 '입력 B 갈래 -> 74HC08 pin 2', width=2, marker_at=(r['b'] + 14, L(8)))
    # Ci/Bi 갈래: 좌 i8 → 3행 → 우 i3 → 우 i11 (74HC08 pin 13, AND2)
    bb.step_wire([(c['i'], L(8)), (c['i'], L(3)), (r['i'], L(3)), (r['i'], L(11))], 'sig',
                 f'입력 {cin} 갈래 -> 74HC08 pin 13 (AND2)', width=2, marker_at=(r['i'] + 14, L(6)))

    if sub:
        # --- 좌 S2: 74HC04. NOT1 = 1→2(a쪽), NOT2 = 게이트6 13→12(f쪽) ---
        bb.draw_chip('74HC04', _S2)
        bb.step_wire([(c['d'], L(7)), (c['d'], L(_S2))], 'sig',
                     '입력 A 갈래 -> 74HC04 pin 1', width=2, marker_at=(c['d'] + 16, L(15)))
        bb.step_wire([(c['c'], L(_S2 + 1)), (c['c'], L(17)), (_LANE_A, L(17)), (_LANE_A, L(10)), (r['a'], L(10))],
                     'mid', "A' (74HC04 pin 2) -> 74HC08 pin 1", marker_at=(_LANE_A - 14, L(13)))
        # A⊕B → NOT2 pin 13 (f쪽 19행): a9 → 16행 → h16 → h19
        bb.step_wire([(c['a'], L(9)), (c['a'], L(16)), (c['h'], L(16)), (c['h'], L(_S2 + 1))], 'mid',
                     'A⊕B (74HC86 pin 3) -> 74HC04 pin 13 (NOT2)', marker_at=(c['a'] - 20, L(13)))
        # (A⊕B)' = pin 12 (f쪽 20행) → 74HC08 pin 12: i20 → 25행 → 452 차선 → 4행 → 우 h4 → h12
        # 4행은 전가산기의 A 갈래 자리였다 — 걷기 보드에서 걷혔으므로 여기선 비어 있다
        bb.step_wire([(c['i'], L(_S2 + 2)), (c['i'], L(25)), (_LANE_X, L(25)), (_LANE_X, L(4)),
                      (r['h'], L(4)), (r['h'], L(12))], 'mid',
                     "(A⊕B)' (74HC04 pin 12) -> 74HC08 pin 12 (AND2)", marker_at=(_LANE_X + 14, L(20)))
    else:
        # A⊕B → AND2 pin 12 (우 f쪽 12행): a9 → 6행 → 우 h6 → h12. 전감산기에서 걷는 가닥
        bb.step_wire([(c['a'], L(9)), (c['a'], L(6)), (r['h'], L(6)), (r['h'], L(12))], 'mid',
                     'A⊕B (74HC86 pin 3) -> 74HC08 pin 12 (AND2)', marker_at=(r['h'] + 14, L(8)))

    # --- 우 S4: 74HC32 게이트1 ---
    bb.draw_chip('74HC32', _S4, block='right')
    bb.step_wire([(r['d'], L(12)), (r['d'], L(_S4 + 1))], 'mid',
                 'AND1 출력 (74HC08 pin 3) -> 74HC32 pin 2', marker_at=(r['d'] + 16, L(18)))
    # AND2 출력 pin 11 (f쪽 13행) → OR pin 1 (a쪽 21행): g13 → 17행(칩 아래) → b17 → b21
    bb.step_wire([(r['g'], L(13)), (r['g'], L(17)), (r['b'], L(17)), (r['b'], L(_S4))], 'mid',
                 'AND2 출력 (74HC08 pin 11) -> 74HC32 pin 1', marker_at=(r['b'] - 16, L(19)))
    bb.step_wire([(r['c'], L(_S4 + 2)), (r['c'], L(_LED2))], 'sig',
                 f'{carry_tag} = 74HC32 pin 3 -> LED2', marker_at=(r['c'] - 16, L(26)))
    carry_desc = {'fa': 'Co = A·B + (A⊕B)·Ci — 진리표 T3 의 Co 열',
                  'fs': "Bo = A'·B + (A⊕B)'·Bi — 진리표 T7 의 Bo 열"}[circuit]
    bb.led_chain(r['c'], _LED2, carry_tag, carry_desc, block='right')
    bb.connect_power(_S4, block='right')
    if sub:
        bb.connect_power(_S2, gnd_col='a')
```

```python
def side_by_side(left_path, right_path, out_path, crop_h=900):
    """두 그림의 보드 부분(위 crop_h px)만 나란히. 왼쪽 = 승인본, 오른쪽 = 시험."""
    a, b = Image.open(left_path).convert('RGB'), Image.open(right_path).convert('RGB')
    a, b = a.crop((0, 0, a.width, min(crop_h, a.height))), b.crop((0, 0, b.width, min(crop_h, b.height)))
    gap = 40
    im = Image.new('RGB', (a.width + gap + b.width, max(a.height, b.height) + 30), 'white')
    im.paste(a, (0, 30)); im.paste(b, (a.width + gap, 30))
    d = ImageDraw.Draw(im)
    d.text((10, 8), 'approved (gates 1,2)', fill='black')
    d.text((a.width + gap + 10, 8), 'trial (gates 1,4 / 1,6)', fill='black')
    im.save(out_path)
```
