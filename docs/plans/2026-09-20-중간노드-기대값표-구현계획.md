**Goal:** 결선도 회로 절의 `## 노드` 표(노드 · 식 · 핀)에서 입력 행마다 모든 게이트 출력 핀의 기대값을 계산해 검증하고, 체크리스트 회로 페이지의 기록표 바로 위에 "프로브로 찍을 핀" 표로 인쇄한다.

**Tech Stack:** Python 3.11 표준 라이브러리만. 테스트는 `unittest` 스타일을 `pytest` 로 돈다 (`python -m pytest`). PDF 재생성은 헤드리스 Chrome (`scripts/lab/html2pdf.py`).

```python
def _split_subtable(body, head_re, columns, keys, label, path, line):
    """회로 본문에서 `## <label>` 헤딩과 그 표를 떼어 `(행 dict 목록, 나머지 본문)` 으로.

    떼어내는 이유: 남겨 두면 `render._circuit_body` 가 "칩이 아닌 헤딩 아래의
    표" 규칙으로 참조표로 인쇄한다 — 블록과 표가 두 번 찍힌다.
    `## 단계` 와 `## 노드` 가 같은 규칙이라 한 함수다.
    """
    lines = body.split("\n")
    for i, ln in enumerate(lines):
        if not head_re.match(ln.rstrip()):
            continue
        j = i + 1
        while j < len(lines) and not lines[j].strip():
            j += 1
        table, nxt = md.parse_table_at(lines, j) if j < len(lines) else (None, j)
        if table is None:
            raise NetlistError(
                f"{path}:{line + i} 근처 `## {label}` 아래에 표가 없다. "
                f"`| {' | '.join(columns)} |` 표를 헤딩 바로 아래 둔다")
        if tuple(h.strip() for h in table.header) != columns:
            raise NetlistError(
                f"{path}:{line + i} 근처 `## {label}` 표의 열이 {table.header} 다. "
                f"{list(columns)} 이어야 한다 — 렌더러가 열 이름으로 읽는다")
        rows = [dict(zip(keys, (c.strip().strip("`") for c in row)))
                for row in table.rows]
        if not rows:
            raise NetlistError(f"{path}:{line + i} 근처 `## {label}` 표에 행이 없다")
        return rows, "\n".join(lines[:i] + lines[nxt:])
    return [], body


def _split_stages(body, path, line):
    """`## 단계` 표. 실험01 실험실 기록(2026-09-11): 복합회로를 전부 꽂고 불이 안
    들어오자 어디가 틀렸는지 알 수 없었다. 칩 하나마다 LED 로 확인하는 단계가
    데이터여야 체크리스트에 확인 칸이 생긴다."""
    return _split_subtable(body, _STAGE_HEAD, STAGE_COLUMNS,
                           ("n", "diagram", "todo", "check"), "단계", path, line)


def _split_nodes(body, path, line):
    """`## 노드` 표. 단계 끝에서 LED 만 보면 중간단은 관측점이 없다 — 프로브로 찍을
    핀과 그 기대값 식을 데이터로 둬야 실패한 행에서 고장 노드를 좁힐 수 있다."""
    return _split_subtable(body, _NODE_HEAD, NODE_COLUMNS,
                           ("name", "expr", "pin"), "노드", path, line)
```

```python
"""`## 노드` 표 × 기록 진리표 → 회로 안 관측점(게이트 출력 핀)의 기대값.

용의자를 "선"이 아니라 "노드"로 좁히는 표다. 틀린 입력 행에서 입력 쪽 핀부터
프로브로 찍어 기대값과 처음 갈라지는 노드가 고장 자리이고, 용의자는 그 게이트와
그 게이트로 들어오는 두세 가닥뿐이다 (2026-09-20 디버깅 회고,
docs/plans/2026-09-20-중간노드-기대값표-설계.md).

이 모듈은 `spec_parser`·`boolexpr`·`notation` 만 import 한다 — `validate` 가
`render` 를 import 하므로 둘이 공유하는 계산은 여기에 있어야 순환이 없다.
"""

from dataclasses import dataclass

from .boolexpr import eval_boolean
from .notation import bit_of, normalize_cell
from .spec_parser import split_indices


@dataclass
class NodeTable:
    inputs: list      # 입력 열 이름 (기록표 헤더 순서)
    nodes: list       # (노드 이름, 핀) — `## 노드` 표 순서
    rows: list        # 행마다 (입력 셀 문자열 리스트, 노드 비트 리스트 | None)
    encoding: str     # 기록표의 encoding — 셀은 이것으로 찍는다 (H/L ↔ 0/1 변환 금지)


def record_table(circuit, spec):
    """회로 `tables` 의 첫 기록 진리표. 없으면 None.

    둘 이상이면 입력 열이 첫 표와 같아야 한다 — 다른 입력 집합을 한 노드 표에 섞을 수
    없다. `role: reference` 는 교안 이론표라 제외하고, `stepped`·`capture` 는 입력 행이
    없어 제외한다.
    """
    tables = [spec.table(t) for t in circuit.tables]
    tables = [t for t in tables
              if t is not None and t.role != "reference" and t.kind == "truth-table"]
    if not tables:
        return None
    first = tables[0]
    inputs = [first.header[i] for i in split_indices(first)[0]]
    for t in tables[1:]:
        other = [t.header[i] for i in split_indices(t)[0]]
        if other != inputs:
            raise ValueError(
                f"{t.id} 의 입력 열 {other} 이 {first.id} 의 {inputs} 와 다르다 — "
                "노드 표는 입력 열이 같은 기록표에만 걸친다")
    return first


def compute(circuit, spec) -> NodeTable:
    """기록표의 입력 행마다 `## 노드` 표의 식을 전부 평가한다.

    입력에 don't-care 가 있는 행은 구체적 식으로 계산할 수 없어 `None` —
    `validate.check_boolean` 이 그런 행을 건너뛰는 것과 같은 이유다.
    """
    table = record_table(circuit, spec)
    if table is None:
        raise ValueError(
            f"{circuit.id} 에 truth-table 기록표가 없다 — 노드 표는 입력 행이 있는 회로에만 둔다")
    inputs, _ = split_indices(table)
    names = [table.header[i] for i in inputs]
    rows = []
    for row in table.rows:
        if len(row) != len(table.header):
            continue              # 셀 수 불일치는 check_table_integrity 몫이다
        cells = [str(normalize_cell(row[i])) for i in inputs]
        env = {}
        for name, i in zip(names, inputs):
            bit = bit_of(row[i], table.encoding)
            if bit is None:
                env = None
                break
            env[name] = bit
        if env is None:
            rows.append((cells, None))
            continue
        bits = []
        for node in circuit.nodes:
            try:
                bits.append(eval_boolean(node["expr"], env))
            except ValueError as exc:
                raise ValueError(f"노드 {node['name']} 의 식 {node['expr']!r}: {exc}") from exc
        rows.append((cells, bits))
    return NodeTable(inputs=names,
                     nodes=[(n["name"], n["pin"]) for n in circuit.nodes],
                     rows=rows, encoding=table.encoding or "0/1")


def format_bit(bit, encoding) -> str:
    """인쇄용. 표의 encoding 을 따른다 — 변환이 아니라 표기다."""
    if bit is None:
        return "—"
    if encoding == "H/L":
        return "H" if bit else "L"
    return "1" if bit else "0"

def _check_nodes(circuit, spec, where) -> list:
    """`## 노드` 표 — 식·핀·출력 노드. 용의자를 '선'이 아니라 '노드'로 좁히는 표라

    출력 노드(명세 `derived` 의 대입 대상)는 반드시 노드 표에 있어야 하고, 그 식의
    값이 명세 식과 전 행 일치해야 한다 — 결선도와 명세의 표류를 잡는 유일한 자리다.
    `check_boolean` 은 명세 셀만 본다.
    """
    out = []
    if not circuit.nodes:
        if circuit.stages:
            out.append(Finding("warn", where,
                "단계는 있는데 `## 노드` 표가 없다 — 단계 끝을 프로브로 찍을 핀과 "
                "기대값이 없다. 게이트 출력 핀마다 `| 노드 | 식 | 핀 |` 한 행을 적는다"))
        return out
    try:
        table = nodes.record_table(circuit, spec)
    except ValueError as exc:
        return [Finding("error", where, str(exc))]
    if table is None:
        return [Finding("error", where,
            "`## 노드` 표는 truth-table 기록표를 가진 회로에만 둔다 — "
            "입력 행이 있어야 기대값이 있다")]
    declared = {u for u, _ in IC_DECL.findall(circuit.body)}
    seen = set()
    for node in circuit.nodes:
        if node["name"] in seen:
            out.append(Finding("error", where, f"노드 {node['name']!r} 이 두 번 있다"))
        seen.add(node["name"])
        if not _NODE_PIN.match(node["pin"]):
            out.append(Finding("error", where,
                f"노드 {node['name']} 의 핀 {node['pin']!r} 은 `U<n> pin <m>` 형식이 아니다"))
        elif node["pin"].split()[0] not in declared:
            out.append(Finding("error", where,
                f"노드 {node['name']} 의 칩 {node['pin'].split()[0]} 이 이 회로에 "
                f"선언돼 있지 않다. 선언된 칩: {sorted(declared)}"))
    try:
        computed = nodes.compute(circuit, spec)
    except ValueError as exc:
        out.append(Finding("error", where, str(exc)))
        return out
    by_name = {name: k for k, (name, _) in enumerate(computed.nodes)}
    for target, expr in derived_parts(table):
        if target not in by_name:
            out.append(Finding("error", where,
                f"출력 {target} 이 `## 노드` 표에 없다 — LED 가 달린 핀도 노드다"))
            continue
        k = by_name[target]
        for n, (cells, bits) in enumerate(computed.rows, start=1):
            if bits is None:
                continue
            env = {name: _bit(c, table.encoding) for name, c in zip(computed.inputs, cells)}
            want = eval_boolean(expr, env)
            if bits[k] != want:
                shown = ", ".join(f"{i}={c}" for i, c in zip(computed.inputs, cells))
                out.append(Finding("error", f"{where} 행{n}",
                    f"{shown} 일 때 노드 {target} = {bits[k]} 인데 명세 식 {expr} 은 "
                    f"{want} 다 — 결선도와 명세가 표류했다"))
                break
    return out

def _node_table_html(spec, section) -> str:
    """중간 노드 기대값 — 프로브로 찍을 핀. 기록표 바로 위에 온다.

    `.check` 셀을 두지 않는다 — `.check` 개수는 명세의 truth-table 행 수와 대조되는
    불변식이다(`test_check_cells_exist_for_every_truth_table_row`). 이 표는 채우는
    표가 아니라 읽는 표다. 2026-09-20 회고: 다 꽂고 켰을 때 틀리면 용의자가 선
    전부였다 — 노드 하나를 찍을 때마다 절반으로 준다.

    계산 실패는 validate 가 먼저 error 로 막지만, 렌더는 죽지 않고 안내 한 줄을 낸다.
    """
    if not section.nodes:
        return ""
    try:
        nt = nodes.compute(section, spec)
    except ValueError as exc:
        return f"<p class='src'>노드 표를 계산할 수 없다: {_esc(exc)}</p>"
    out = ['<div class="tblock nodes">',
           "<h3>중간 노드 기대값 — 프로브로 찍을 핀</h3>",
           f"<p class='src'>{_esc(_NODE_HINT)}</p>",
           "<table>", "<thead>", "<tr>"]
    out += [f"<th>{_esc(name)}</th>" for name in nt.inputs]
    out += [f"<th>{_esc(_plain(name))}<br><small>{_esc(pin)}</small></th>"
            for name, pin in nt.nodes]
    out += ["</tr>", "</thead>", "<tbody>"]
    for cells, bits in nt.rows:
        vals = ([nodes.format_bit(b, nt.encoding) for b in bits] if bits is not None
                else ["—"] * len(nt.nodes))
        out.append("<tr>" + "".join(f"<td>{_esc(c)}</td>" for c in cells)
                   + "".join(f"<td>{_esc(v)}</td>" for v in vals) + "</tr>")
    out += ["</tbody>", "</table>", "</div>"]
    return "\n".join(out)
```
