# 체크리스트 강의노트 회로도 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 명세 항목에 `schematic:` 필드를 더해, 체크리스트 회로 페이지 첫머리(칩 교체 배너 뒤)에 강의노트 게이트 회로도 크롭 PNG 를 싣는다.

**Architecture:** 파서가 항목의 `schematic`(명세 기준 상대경로 리스트)을 읽고, `validate.check_schematics` 가 파일 부재를 warn 으로 알리고, `render.circuit_pages` 가 항목의 첫 회로에 `<img class="schematic">` 을 낸다. PNG 는 새 모듈 `scripts/lab/schematic.py` 가 교안 PDF 를 pymupdf 로 300dpi 렌더·크롭해 `<실험NN>/체크리스트/` 에 저장하고 명세에 필드를 적는다. `draft-lab-spec` 4단계 끝에서 부른다.

**Tech Stack:** Python 3.11, pymupdf(`import fitz`, 1.26), pytest 8.4. 저장소 루트에서 `python -m pytest`.

**Spec:** `docs/plans/2026-09-19-체크리스트-회로도-설계.md`

```python
class TestSchematics(unittest.TestCase):
    """명세 항목 `schematic` 이 가리키는 강의노트 회로도가 명세 옆에 있는가."""

    def _spec(self, schematic_line, make_png):
```

```python
def check_schematics(spec) -> list:
    """항목 `schematic` 이 가리키는 강의노트 회로도 PNG 가 명세 옆에 있는가.

    필드가 없으면 아무것도 내지 않는다 — 실험01·02 는 회로도를 소급하지 않는다.
    파일이 없으면 warn 이다(결선도 `diagram` 부재와 같은 급): 렌더러는 `<img>` 대신
    안내 한 줄을 내므로 인쇄는 가능하다. 경로는 명세 파일 위치 기준이다.
    """
    out = []
    for item in spec.items:
        for rel in item.schematic or []:
            if not (spec.path.parent / rel).exists():
                out.append(Finding(
                    "warn", f"{spec.path.name}:{item.id}",
                    f"회로도 파일이 없다: {rel} — "
                    f"`python -m scripts.lab.schematic {spec.experiment} {item.id}` 로 만든다"))
    return out
```

```python
자리에 4pt 여백을 둔 것이다 — 여백을 더 주면 이미지 바로 아래 캡션 '회로 구성'(p5·p7
top 507pt)이 반쯤 걸린다. 그림 자리가 다르면 `--box` 로 준다.

PNG 는 `<실험NN>/체크리스트/` 에 둔다. `결선도/` 는 `배치.py` 출력만 담는다
(`tests/test_breadboard_boards.py` 의 `CommittedPngs`).

pymupdf(`fitz`)는 함수 안에서만 부른다 — `render`·`validate` 가 이 모듈을 안 부르지만,
fitz 없는 기계에서 `scripts.lab` 을 import 하다 죽는 일을 만들지 않는다.

import argparse
import re
from collections import Counter
from pathlib import Path

from . import config
from .figure import sub_id
from .spec_parser import parse_spec

DPI = 300
MARGIN_PT = 4


def schematic_name(exp_dir, item) -> str:
    """`실험03` + `E3.P1-2` → `실험03_1-2_회로도.png`."""
    return f"{Path(exp_dir).name}_{sub_id(item)}_회로도.png"


def default_box(manual, page: int) -> tuple:
    """그 쪽에서 로고를 뺀 가장 큰 이미지 자리 + 여백 (pt, 쪽 왼쪽 위 기준).

    로고 = 문서 쪽의 절반 이상에 같은 xref 로 나오는 이미지.
    """
    import fitz
    with fitz.open(str(manual)) as doc:
        counts = Counter()
        for p in doc:
            for xref in {info[0] for info in p.get_images(full=True)}:
                counts[xref] += 1
        pg = doc[page - 1]
        rects = [r for xref in {info[0] for info in pg.get_images(full=True)}
                 if counts[xref] * 2 < len(doc)
                 for r in pg.get_image_rects(xref)]
        if not rects:
            raise ValueError(f"{manual} p{page} 에 회로도로 볼 이미지가 없다 — "
                             "--box x0,y0,x1,y1 (pt) 로 영역을 준다")
        r = max(rects, key=lambda r: r.width * r.height)
        box = fitz.Rect(r.x0 - MARGIN_PT, r.y0 - MARGIN_PT,
                        r.x1 + MARGIN_PT, r.y1 + MARGIN_PT) & pg.rect
        return (box.x0, box.y0, box.x1, box.y1)


def crop(manual, page: int, box: tuple, out) -> None:
    import fitz
    with fitz.open(str(manual)) as doc:
        pix = doc[page - 1].get_pixmap(dpi=DPI, clip=fitz.Rect(*box))
        pix.save(str(out))


_FRONT = re.compile(r"\A---\n(.*?)\n---\n", re.S)


def set_spec_schematic(spec_text: str, item_id: str, rel_paths: list) -> str:
    """프론트매터의 그 항목에 `    schematic: [...]` 를 넣거나 바꾼다.

    자리는 `pages:` 다음 줄, 없으면 항목 블록의 끝이다. 항목은 `  - id:`(2칸),
    필드는 4칸 들여쓰기라고 본다 — `draft-lab-spec` 이 쓰는 형태다.
    """
    front = _FRONT.match(spec_text)
    if front is None:
        raise ValueError("명세에 프론트매터가 없다")
    lines = front.group(1).split("\n")
    start = next((i for i, l in enumerate(lines)
                  if re.fullmatch(rf"  - id: {re.escape(item_id)}\s*", l)), None)
    if start is None:
        raise ValueError(f"명세 프론트매터에 항목 {item_id} 가 없다")
    end = next((i for i in range(start + 1, len(lines))
                if not lines[i].startswith("    ")), len(lines))
    new = f"    schematic: [{', '.join(rel_paths)}]"
    block = lines[start:end]
    at = next((i for i, l in enumerate(block) if l.startswith("    schematic:")), None)
    if at is not None:
        block[at] = new
    else:
        pages = next((i for i, l in enumerate(block) if l.startswith("    pages:")), None)
        block.insert(len(block) if pages is None else pages + 1, new)
    lines[start:end] = block
    return f"---\n{chr(10).join(lines)}\n---\n" + spec_text[front.end():]


def main(argv=None) -> int:
    ap = argparse.ArgumentParser(
        description="강의노트 회로도를 300dpi 로 크롭해 체크리스트/ 에 두고 명세 schematic 에 적는다")
    ap.add_argument("experiment", type=int)
    ap.add_argument("item", help="명세 항목 id (예: E3.P1-2)")
    ap.add_argument("--page", type=int, help="교안 쪽. 없으면 항목 pages 의 첫 쪽")
    ap.add_argument("--box", help="x0,y0,x1,y1 (pt, 쪽 왼쪽 위 기준). 없으면 default_box")
    a = ap.parse_args(argv)

    root = config.find_project_root(Path("."))
    exp_dir = root / config.experiment_dir(a.experiment)
    spec_path = exp_dir / "체크리스트/명세.md"
    spec = parse_spec(spec_path)
    item = next((i for i in spec.items if i.id == a.item), None)
    if item is None:
        ap.error(f"명세에 항목 {a.item} 가 없다: {[i.id for i in spec.items]}")
    page = a.page or (item.pages[0] if item.pages else None)
    if page is None:
        ap.error(f"항목 {item.id} 에 pages 가 없다 — --page 로 쪽을 준다")
    manual = root / spec.source_manual
    if a.box:
        box = tuple(float(v) for v in a.box.split(","))
        if len(box) != 4:
            ap.error("--box 는 x0,y0,x1,y1 네 수다")
    else:
        box = default_box(manual, page)
    name = schematic_name(exp_dir, item)
    out = spec_path.parent / name
    crop(manual, page, box, out)
    spec_path.write_text(
        set_spec_schematic(spec_path.read_text(encoding="utf-8"), item.id, [name]),
        encoding="utf-8")
    print(f"{out} ← {spec.source_manual} p{page} box={tuple(round(v, 1) for v in box)}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
def _schematic_html(item, spec_dir) -> str:
    """강의노트 게이트 회로도 — 항목의 첫 회로 페이지, 칩 교체 배너 바로 뒤.

    결선도를 보다가 "원래 회로가 뭐였지" 를 교안을 따로 열지 않고 확인하게 한다.
    `src` 는 명세 기준 상대경로 그대로다 — 인쇄본 HTML 이 명세와 같은 `체크리스트/`

    내지 않는다 — 깨진 그림이 PDF 로 굳는다. 필드가 없으면 빈 문자열이다.
    """
    if item is None or not item.schematic:
        return ""
    pages = "·".join(f"p{p}" for p in item.pages or [])
    caption = f"강의노트 {pages} · 회로 구성" if pages else "강의노트 · 회로 구성"
    out = []
    for rel in item.schematic:
        if (Path(spec_dir) / rel).exists():
            out.append(f'<img class="schematic" src="{_esc(rel)}">')
        else:
            out.append(f"<p class='src'>회로도가 아직 없다: {_esc(rel)}</p>")
    out.append(f"<p class='src'>{_esc(caption)}</p>")
    return "".join(out)
```

```python
    out, previous = [], None
    shown = set()                 # 강의노트 회로도를 이미 실은 항목

        tables = ", ".join(section.tables)
        item = next((i for i in spec.items if i.id == section.item), None)
        device = f" · 판독: {', '.join(item.device)}" if item else ""
        out.append('<section class="circuit">')
        out.append(f"<h2>회로 {n}/{total} · {_esc(_plain(section.title))} "
                   f"· 기록: {_esc(tables)}{_esc(device)}</h2>")
        banner = _chip_swap(previous, section)
        out.append(f'<div class="banner">{_esc(banner)}</div>')
        if item is not None and item.id not in shown:
            shown.add(item.id)
            schematic = _schematic_html(item, spec.path.parent)
            if schematic:         # 빈 문자열을 넣으면 필드 없는 주차의 HTML 에 빈 줄이 는다
```
