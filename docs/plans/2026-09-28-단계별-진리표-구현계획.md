# 단계별 진리표 구현계획

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 체크리스트 회로 페이지의 관측 단계마다 그 노드의 기대값 진리표(단계 진리표)를 단계 블록 안에 싣고, 기록표를 마지막 단계 바로 뒤로 옮기며, 회로 끝의 중간 노드 기대값 표를 없앤다.

**Architecture:** `boolexpr.variables` 가 노드 식의 입력 변수를 뽑고, `nodes.stage_table` 이 명세 기록표 행을 그 변수로 투영·중복 제거하며 don't-care 를 풀어 노드 값을 정한다. `render._stage_html` 이 그 결과를 `stage-table` 로 찍고, `render.circuit_pages` 가 순서를 `완성본·배선 → 핀 점검 → 단계 → 기록표` 로 바꾼다. `validate` 가 `render` 를 import 하므로 계산은 `nodes.py` 에 둔다.

**Tech Stack:** Python 3.11 표준 라이브러리만. 테스트는 `unittest` 스타일을 `pytest` 로 돈다(`.venv/bin/python -m pytest`). PDF 는 헤드리스 Chrome (`scripts/lab/html2pdf.py`).

**Spec:** `docs/plans/2026-09-28-단계별-진리표-설계.md`

그 게이트로 들어오는 두세 가닥뿐이다 (2026-09-20 디버깅 회고,
docs/plans/2026-09-20-중간노드-기대값표-설계.md).

`compute` 는 모든 노드 × 기록표 전 행이다 — `validate` 가 출력 노드를 명세 식과
대조하는 데 쓴다. `stage_table` 은 노드 하나 × 그 노드가 쓰는 입력만이다 — 체크리스트

이 모듈은 `spec_parser`·`boolexpr`·`notation` 만 import 한다 — `validate` 가
`render` 를 import 하므로 둘이 공유하는 계산은 여기에 있어야 순환이 없다.
"""

from dataclasses import dataclass
from itertools import product

from .boolexpr import eval_boolean, variables
from .notation import DONT_CARE, bit_of, normalize_cell
from .spec_parser import split_indices


@dataclass
class NodeTable:
    inputs: list      # 입력 열 이름 (기록표 헤더 순서)
    nodes: list       # (노드 이름, 핀) — `## 노드` 표 순서
    rows: list        # 행마다 (입력 셀 문자열 리스트, 노드 비트 리스트 | None)
    encoding: str     # 기록표의 encoding — 셀은 이것으로 찍는다 (H/L ↔ 0/1 변환 금지)


@dataclass
class StageTable:
    inputs: list      # 이 노드 식에 나오는 입력 열 이름 (기록표 헤더 순서)
    node: tuple       # (노드 이름, 핀)
    rows: list        # 행마다 (입력 셀 문자열 리스트, 노드 비트 0/1) — 값이 정해진 행만
    encoding: str     # 기록표의 encoding


def record_table(circuit, spec):
    """회로 `tables` 의 첫 기록 진리표. 없으면 None.

    둘 이상이면 입력 열이 첫 표와 같아야 한다 — 다른 입력 집합을 한 노드 표에 섞을 수
    없다. `role: reference` 는 교안 이론표라 제외하고, `stepped`·`capture` 는 입력 행이
    없어 제외한다.
    """
    tables = [spec.table(t) for t in circuit.tables]
    tables = [t for t in tables
              if t is not None and t.role != "reference" and t.kind == "truth-table"]
    if not tables:
        return None
    first = tables[0]
    inputs = [first.header[i] for i in split_indices(first)[0]]
    for t in tables[1:]:
        other = [t.header[i] for i in split_indices(t)[0]]
        if other != inputs:
            raise ValueError(
                f"{t.id} 의 입력 열 {other} 이 {first.id} 의 {inputs} 와 다르다 — "
                "노드 표는 입력 열이 같은 기록표에만 걸친다")
    return first


def compute(circuit, spec) -> NodeTable:
    """기록표의 입력 행마다 `## 노드` 표의 식을 전부 평가한다.

    입력에 don't-care 가 있는 행은 구체적 식으로 계산할 수 없어 `None` —
    `validate.check_boolean` 이 그런 행을 건너뛰는 것과 같은 이유다.
    """
    table = record_table(circuit, spec)
    if table is None:
        raise ValueError(
            f"{circuit.id} 에 truth-table 기록표가 없다 — 노드 표는 입력 행이 있는 회로에만 둔다")
    inputs, _ = split_indices(table)
    names = [table.header[i] for i in inputs]
    rows = []
    for row in table.rows:
        if len(row) != len(table.header):
            continue              # 셀 수 불일치는 check_table_integrity 몫이다
        cells = [str(normalize_cell(row[i])) for i in inputs]
        env = {}
        for name, i in zip(names, inputs):
            bit = bit_of(row[i], table.encoding)
            if bit is None:
                env = None
                break
            env[name] = bit
        if env is None:
            rows.append((cells, None))
            continue
        bits = []
        for node in circuit.nodes:
            try:
                bits.append(eval_boolean(node["expr"], env))
            except ValueError as exc:
                raise ValueError(f"노드 {node['name']} 의 식 {node['expr']!r}: {exc}") from exc
        rows.append((cells, bits))
    return NodeTable(inputs=names,
                     nodes=[(n["name"], n["pin"]) for n in circuit.nodes],
                     rows=rows, encoding=table.encoding or "0/1")


def _settled(expr, fixed, free):
    """`free` 변수에 0·1 을 다 넣어도 값이 하나면 그 값, 갈리면 None."""
    values = {eval_boolean(expr, {**fixed, **dict(zip(free, combo))})
              for combo in product((0, 1), repeat=len(free))}
    return values.pop() if len(values) == 1 else None


def stage_table(circuit, spec, node_name) -> StageTable:
    """단계 하나에서 볼 노드의 진리표 — 입력은 그 노드 식에 나오는 것만.

    행은 기록표 행을 그 입력으로 줄여 겹치는 것을 합친 것이다(처음 나온 순서). 2^k 로
    전개하지 않는다 — 실험04 MUX 는 입력이 7개이고, 기록표가 저작자가 고른 시험 행이다.
    줄인 입력에 X 가 남으면 0·1 을 다 넣어 보고 값이 하나로 정해질 때만 X 째 싣는다
    (MUX `E=1` 행은 NAND 출력이 1 로 정해진다). 0/1(H/L)도 X 도 아닌 셀의 행은 뺀다.

    입력 열은 이름이 아니라 `split_indices` 의 **자리**로 고른다 — 입력과 출력이 같은
    글자인 표가 있다(실험3 반감산기 `A B D B`).
    """
    table = record_table(circuit, spec)
    if table is None:
        raise ValueError(
            f"{circuit.id} 에 truth-table 기록표가 없다 — 단계 진리표는 입력 행이 있는 회로에만 둔다")
    node = next((n for n in circuit.nodes if n["name"] == node_name), None)
    if node is None:
        raise ValueError(f"노드 {node_name!r} 가 {circuit.id} 의 `## 노드` 표에 없다")
    inputs, _ = split_indices(table)
    names = [table.header[i] for i in inputs]
    try:
        used = variables(node["expr"])
    except ValueError as exc:
        raise ValueError(f"노드 {node['name']} 의 식 {node['expr']!r}: {exc}") from exc
    missing = [v for v in used if v not in names]
    if missing:
        raise ValueError(f"노드 {node['name']} 의 식 {node['expr']!r}: "
                         f"입력 변수 {missing} 이(가) 기록표 {table.id} 에 없다")
    cols = [(name, i) for name, i in zip(names, inputs) if name in used]
    rows, seen = [], set()
    for row in table.rows:
        if len(row) != len(table.header):
            continue              # 셀 수 불일치는 check_table_integrity 몫이다
        cells = [str(normalize_cell(row[i])) for _, i in cols]
        if tuple(cells) in seen:
            continue
        seen.add(tuple(cells))
        fixed, free, settable = {}, [], True
        for (name, i), cell in zip(cols, cells):
            bit = bit_of(row[i], table.encoding)
            if bit is not None:
                fixed[name] = bit
            elif cell == DONT_CARE:
                free.append(name)
            else:
                settable = False
        if not settable:
            continue
        try:
            bit = _settled(node["expr"], fixed, free)
        except ValueError as exc:
            raise ValueError(f"노드 {node['name']} 의 식 {node['expr']!r}: {exc}") from exc
        if bit is not None:
            rows.append((cells, bit))
    return StageTable(inputs=[name for name, _ in cols],
                      node=(node["name"], node["pin"]),
                      rows=rows, encoding=table.encoding or "0/1")


def format_bit(bit, encoding) -> str:
    """인쇄용. 표의 encoding 을 따른다 — 변환이 아니라 표기다."""
    if bit is None:
        return "—"
    if encoding == "H/L":
        return "H" if bit else "L"
    return "1" if bit else "0"
```

def _stage_html(stage, netlist, section, spec=None, last=False) -> str:
    """단계 블록 하나 — 그림, 이 단계에서 할 것, 확인 노드, 확인 칸, 단계 진리표.

    그림이 없으면 빈 브레드보드를 넣지 않는다 — 단계 그림은 회색 기준선이 요점이라
    격자로 대신할 수 없다. 부재는 validate 가 warn 으로 알렸다.

    `확인 노드` 줄은 그 단계에서 볼 노드의 이름과 **핀**이다. 핀은 `## 노드` 표에서
    조회한다 — 단계 표에 핀을 적게 하면 두 자리가 표류한다.

    단계 진리표는 그 노드의 기대값을 **이 블록 안에** 싣는다 — 2026-09-28 전에는 회로 끝의
    중간 노드 표에만 있어 학생이 단계마다 뒤로 넘겨 보고 돌아와야 했다. `spec` 이 없으면
    (단위 테스트의 세 인자 호출) 싣지 않는다. `last` 인 단계는 싣지 않는다 — 바로 뒤에
    기록표가 오고 그것이 그 단계의 표다.

    걷기 행(`<k>걷기`)에는 `확인 노드` 줄도 단계 진리표도 없다 — 걷기만 하는 장이라 관측할
    출력이 없고, 파서가 `node` 를 `None` 으로 낸다.
    """
    png = netlist.path.parent / stage["diagram"] if stage["diagram"] else None
    if png is not None and png.exists():
        src = f"../{netlist.path.parent.name}/{stage['diagram']}"
        img = f'<img src="{_esc(src)}" style="width:100%">'
    else:
        img = f"<p class='src'>단계 그림이 아직 없다: {_esc(stage['diagram'] or '(없음)')}</p>"
    node = ""
    if stage.get("node"):
        pin = next((n["pin"] for n in section.nodes if n["name"] == stage["node"]), "")
        tail = f" <small>({_esc(pin)})</small>" if pin else ""
        node = f"<p><b>확인 노드:</b> {_esc(_plain(stage['node']))}{tail}</p>"
    table = ""
    if spec is not None and stage.get("node") and not last:
        table = _stage_table_html(spec, section, stage["node"])
    # 걷기 행은 `단계 2걷기` 가 아니라 `단계 2 · 걷기` 로 찍는다 — 인쇄본에서
    # "2단계 머리의 걷기" 로 읽혀야 한다. 판정은 파서의 `walk` 키다.
    head = (f"단계 {_esc(str(stage['walk_of']))} · 걷기" if stage.get("walk")
            else f"단계 {_esc(stage['n'])}")
    return (f'<div class="stage"><h3>{head}</h3>{img}'
            f"<p><b>이 단계에서:</b> {_esc(_plain(stage['todo']))}</p>{node}"
            f'<p><span class="box"></span><b>확인:</b> {_esc(_plain(stage["check"]))}</p>'
            f"{table}</div>")


def _stage_table_html(spec, section, node_name) -> str:
    """단계 진리표 — 입력은 그 노드 식에 나오는 것만, 열 머리 둘째 줄은 노드 핀.

    확인 칸은 `td.box` 다 — `.check` 개수는 명세 기록표 행 수와 대조되는 불변식이다
    (`test_check_cells_exist_for_every_truth_table_row`). 표를 div 로 감싸지 않는다 —
    단계 블록의 `</div>` 가 블록 끝이어야 테스트와 인쇄 묶음(`break-inside`)이 맞는다.

    계산 실패는 validate 가 먼저 error 로 막지만, 렌더는 죽지 않고 안내 한 줄을 낸다.
    """
    try:
        st = node_calc.stage_table(section, spec, node_name)
    except ValueError as exc:
        return f"<p class='src'>단계 진리표를 계산할 수 없다: {_esc(exc)}</p>"
    name, pin = st.node
    head = ("<tr>" + "".join(f"<th>{_esc(n)}</th>" for n in st.inputs)
            + f"<th>{_esc(_plain(name))}예상<br><small>{_esc(pin)}</small></th>"
            + "<th>확인</th></tr>")
    body = ["<tr>" + "".join(f"<td>{_esc(c)}</td>" for c in cells)
            + f"<td>{_esc(node_calc.format_bit(bit, st.encoding))}</td>"
            + '<td class="box"></td></tr>'
            for cells, bit in st.rows]
    return "\n".join(['<table class="stage-table">', "<thead>", head, "</thead>",
                      "<tbody>", *body, "</tbody>", "</table>"])
```

- [ ] **Step 5: `circuit_pages` 를 바꾼다**

`def circuit_pages(spec, netlist) -> str:` 함수 전체를 다음으로 바꾼다. 달라진 곳은 docstring 순서 문장, 단계 루프의 자리(핀 점검·뒤 산문 뒤로), `last` 계산, 노드 표 호출 삭제뿐이다.

```python
def circuit_pages(spec, netlist) -> str:
    """회로마다 한 면. 강제 개쪽이라 회로 둘이 한 쪽을 나눠 쓰지 않는다.
    강의노트 회로도는 한 항목을 가리키는 회로가 여럿이어도 첫 회로에만 싣는다.

    면 안의 순서는 작업 순서다 — 칩 교체 → 강의노트 회로도 → 유의사항 → 결선도
    → 배선 → 미사용핀 → 단계(단계 진리표) → 측정표. 유의사항(예: `74HC02 는 핀 순서에 주의`)을
    배선 앞으로 옮겼다 — 그것을 실행하며 봐야 할 배선 뒤에 두면 이미 다 꽂은 뒤에야 읽게 된다.
    단계는 완성본·배선 **뒤**다(2026-09-28) — 전체 모습을 먼저 보고 단계를 밟는다. 측정표는
    마지막 단계 바로 뒤라, 마지막 관측 단계는 자기 단계 진리표 대신 기록표를 본다.
    미사용핀 상자는 칩별로 낸다 — 회로 단위로 합치면 C4 처럼 같은 핀 번호가
    칩마다 정반대 지시를 받을 때 어느 칩 것인지 사라진다.
    """
    out, previous = [], None
    shown = set()                 # 강의노트 회로도를 이미 실은 항목
    total = len(netlist.circuits)
    for n, section in enumerate(netlist.circuits, start=1):
        nodes, prose_before, prose_after = _circuit_body(section)
        chips = _chip_boxes(nodes)
        tables = ", ".join(section.tables)
        item = next((i for i in spec.items if i.id == section.item), None)
        device = f" · 판독: {', '.join(item.device)}" if item else ""
        out.append('<section class="circuit">')
        out.append(f"<h2>회로 {n}/{total} · {_esc(_plain(section.title))} "
                   f"· 기록: {_esc(tables)}{_esc(device)}</h2>")
        banner = _chip_swap(previous, section)
        out.append(f'<div class="banner">{_esc(banner)}</div>')
        if item is not None and item.id not in shown:
            shown.add(item.id)
            schematic = _schematic_html(item, spec.path.parent)
            if schematic:         # 빈 문자열을 넣으면 필드 없는 주차의 HTML 에 빈 줄이 는다
                out.append(schematic)
        if prose_before.strip():
            out.append(_blocks(prose_before))
        out.append('<div class="wirebox">')
        if section.stages:
            out.append("<p class='src'><b>완성본 — 손그림 원도</b> (회색 없음)</p>")
        out.append(_diagram_html(section, netlist))
        for node in nodes:
            if node["type"] == "prose":
                out.append(_blocks(node["text"]))
                continue
            out.append(f'<div class="chip"><h4>{_esc(node["head"])}</h4>')
            if node["wires"]:
                out.append('<ol start="1">')
                out += [f'<li><span class="box"></span>{_esc(w)}</li>'
                        for w in node["wires"]]
                out.append("</ol>")
            # 상자 안은 원문 순서다. 고정 순서로 찍으면 표 앞에 있어야 할 경고가
            # 표 아래로 밀린다 (실험02 U1) — 주차마다 표와 산문의 순서가 다르다.
            for kind, payload in node["blocks"]:
                out.append(_md_table_html(payload, lead_checkbox=True)
                           if kind == "table" else _blocks(payload))
            out.append("</div>")
        out.append("</div>")
        unused_chips = [c for c in chips if c["unused"]]
        if unused_chips:
            out.append('<div class="unused">'
                       "<b>핀 점검 — 사용 핀은 다 물렸는가 · 미사용 핀은 비어 있는가 (칩마다 따로)</b>")
            for chip in unused_chips:
                out.append(f'<div class="chipunused"><b>{_esc(chip["head"])}</b><ul>')
                out += [f'<li><span class="box"></span>{_esc(u)}</li>'
                        for u in chip["unused"]]
                out.append("</ul></div>")
            out.append("</div>")
        if prose_after.strip():
            # 배선을 마친 뒤 하는 검산이다. 앞으로 올리면 preamble 로 읽힌다.
            out.append(_blocks(prose_after))
        # 마지막 관측 단계는 단계 진리표를 싣지 않는다 — 바로 뒤 기록표가 그 표다.
        observed = [s for s in section.stages if s.get("node")]
        last = observed[-1] if observed else None
        for stage in section.stages:
            out.append(_stage_html(stage, netlist, section, spec, last=stage is last))
        out.append(_circuit_tables(spec, section))
        out.append("</section>")
        previous = section
    return "\n".join(out)
```
