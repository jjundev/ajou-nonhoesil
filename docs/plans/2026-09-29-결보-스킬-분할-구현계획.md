    def test_the_listed_siblings_are_exactly_the_md_files_in_the_folder(self):

        self.assertGreaterEqual(checked, 5, "포인터를 못 찾으면 이 테스트는 공허하다")

lines = raw.split(NL)


def one(pred, what):
    hits = [i for i, line in enumerate(lines) if pred(line)]
    if len(hits) != 1:
        sys.exit(f"앵커 {what}: {len(hits)}번")
    return hits[0]


def dedent3(block):
    return [line[3:] if line.startswith("   ") else line for line in block]


# 아래에서 위로 자른다 — 앞쪽 인덱스가 밀리지 않게.
rep_s = one(lambda l: l == "## repeat: per-item 처리", "repeat 절")
rep_e = one(lambda l: l == "## 사진 배치", "사진 배치 절")
quiz_s = one(lambda l: l.startswith("   - **`generate: true` 가 되면:**"), "퀴즈 true")
quiz_e = one(lambda l: l.startswith("9. **md 저장.**"), "9단계")
tr_s = one(lambda l: l == "   PDF 를 읽어 각 표의 확인 칸을 채운다.", "전사 본문")
tr_e = one(lambda l: l.startswith("4. **양식 로드**"), "4단계")
assert tr_s < tr_e < quiz_s < quiz_e < rep_s < rep_e

# 각 덩어리 끝의 빈 줄 하나는 SKILL.md 에 남긴다(다음 항목·절 앞 여백).
repeat_body = lines[rep_s + 1:rep_e - 1]
quiz_body = dedent3(lines[quiz_s:quiz_e - 1])
tr_body = dedent3(lines[tr_s:tr_e - 1])
for name, body in (("repeat", repeat_body), ("quiz", quiz_body), ("tr", tr_body)):
    if any(l.startswith("## ") for l in body):
        sys.exit(f"{name}: 형제 파일에 `## ` 줄이 들어간다")

lines[rep_s:rep_e] = []
lines[quiz_s:quiz_e - 1] = [
    "   - **`generate: true` 면** 이 섹션을 쓰기 전에",
    "     " + P.format("퀴즈해설.md") + " 를 지금 읽고 그대로 따른다.",
]
lines[tr_s:tr_e - 1] = [
    "   전사할 것이 있으면 — 필기 PDF 가 정해졌거나 위 출력에 `빈 칸` 이 하나라도",
    "   있으면 — " + P.format("필기전사.md") + " 를 지금 읽고 그대로",
    "   따른다. 없으면 4단계로 간다.",
]

# 치환표 a~c — 줄 전체가 같을 때만.
subs = {
    "    마찬가지다 — §repeat 처리 절 참고.":
        ["    마찬가지다 — " + P.format("항목반복.md") + " 참고."],
    "   §repeat 처리 절을 따른다.":
        ["   이 단계를 마치기 전에 " + P.format("항목반복.md") + " 를",
         "   지금 읽고 그대로 따른다."],
    "   `repeat: per-item` 프리셋에서는 §repeat 처리 절을 따른다.":
        ["   `repeat: per-item` 프리셋에서는",
         "   " + P.format("항목반복.md") + " 를 따른다."],
}
for old, new in subs.items():
    i = one(lambda l, old=old: l == old, old)
    lines[i:i + 1] = new


def page(title, intro, body):
    return NL.join(["# " + title, "", *intro, "", *body, ""])


files = {
    "필기전사.md": page("필기·사진 전사", [
        "`write-result-report` 실행 절차 3단계에서 전사할 것이 있을 때만 읽는다.",
        "이 파일의 단계·규칙 번호는 `SKILL.md` 의 번호다.",
    ], tr_body),
    "퀴즈해설.md": page("퀴즈 해설 (`generate: true`)", [
        "`write-result-report` 실행 절차 8단계에서 퀴즈 섹션의 `generate` 가",
        "`true` 일 때만 읽는다. 이 파일의 단계·규칙 번호는 `SKILL.md` 의 번호다.",
    ], quiz_body),
    "항목반복.md": page("repeat: per-item 처리", [
        "`write-result-report` 에서 `fmt.result` 에 `repeat == \"per-item\"` 인",

}
for name, text in files.items():
    (DIR / name).write_bytes(text.encode("utf-8"))
SKILL.write_bytes(NL.join(lines).encode("utf-8"))
print("SKILL.md", len(lines), "줄;", {n: t.count(NL) for n, t in files.items()})
