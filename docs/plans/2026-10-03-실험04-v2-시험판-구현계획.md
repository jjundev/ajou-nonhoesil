## Review Focus

1. **칩 라벨 충돌.** 넷 라벨은 `74HC20#<pin1행> pin N` 이라 블록을 담지 않는다. v2 는 칩이 전부 좌 블록이라 행이 겹치지 않는다. 그래도 모의기는 같은 라벨이 두 칩에서 나오면 조용히 섞지 말고 멈춰야 한다 → Task 1 `test_같은_칩_라벨이_두_번이면_멈춘다`.

# tests/test_gate_sim.py
"""게이트 모의기 — 렌더러 넷에서 진리표를 돌린다. 합성 넷만 쓴다(렌더러 불필요)."""
import unittest

from tests._gate_sim import SimError, value_of


def net(*labels, rails=()):
    return {'labels': set(labels), 'rails': set(rails)}


class GateSim(unittest.TestCase):
    def test_인버터는_입력을_뒤집는다(self):
        nets = [net('입력 A', '74HC04#5 pin 1'), net('74HC04#5 pin 2', 'LED X 애노드')]
        self.assertEqual(value_of(nets, 'LED X 애노드', {'A': 0}), 1)
        self.assertEqual(value_of(nets, 'LED X 애노드', {'A': 1}), 0)

    def test_4입력_NAND(self):
        nets = [net('입력 A', '74HC20#17 pin 1'), net('입력 B', '74HC20#17 pin 2'),
                net('74HC20#17 pin 4', '74HC20#17 pin 5', rails=('vcc',)),
                net('74HC20#17 pin 6', 'LED Y 애노드')]
        self.assertEqual(value_of(nets, 'LED Y 애노드', {'A': 1, 'B': 1}), 0)
        self.assertEqual(value_of(nets, 'LED Y 애노드', {'A': 1, 'B': 0}), 1)

    def test_3입력_AND_게이트1은_pin_13_을_입력으로_쓴다(self):
        nets = [net('입력 A', '74HC11#17 pin 1', '74HC11#17 pin 2', '74HC11#17 pin 13'),
                net('74HC11#17 pin 12', 'LED Y0 애노드')]
        self.assertEqual(value_of(nets, 'LED Y0 애노드', {'A': 1}), 1)
        self.assertEqual(value_of(nets, 'LED Y0 애노드', {'A': 0}), 0)

    def test_게이트를_사슬로_잇는다(self):
        nets = [net('입력 A', '74HC04#5 pin 1'),
                net('74HC04#5 pin 2', '74HC04#5 pin 3'),
                net('74HC04#5 pin 4', 'LED Z 애노드')]
        self.assertEqual(value_of(nets, 'LED Z 애노드', {'A': 1}), 1)

    def test_뜬_입력이면_멈춘다(self):
        nets = [net('74HC04#5 pin 1'), net('74HC04#5 pin 2', 'LED X 애노드')]
        with self.assertRaisesRegex(SimError, '뜬'):
            value_of(nets, 'LED X 애노드', {})

    def test_관측과_무관한_게이트의_뜬_입력은_무시한다(self):
        nets = [net('입력 A', '74HC20#41 pin 1'),        # 게이트 A 는 중계로만 쓰고 나머지 입력은 뜬다
                net('74HC20#41 pin 6'),
                net('입력 B', '74HC20#41 pin 9', '74HC20#41 pin 10',
                    '74HC20#41 pin 12', '74HC20#41 pin 13'),
                net('74HC20#41 pin 8', 'LED Y 애노드')]
        self.assertEqual(value_of(nets, 'LED Y 애노드', {'A': 0, 'B': 1}), 0)

    def test_한_넷에_출력이_둘이면_멈춘다(self):
        nets = [net('입력 A', '74HC04#5 pin 1', '74HC04#5 pin 3'),
                net('74HC04#5 pin 2', '74HC04#5 pin 4', 'LED X 애노드')]
        with self.assertRaisesRegex(SimError, '출력 둘'):
            value_of(nets, 'LED X 애노드', {'A': 0})

    def test_같은_칩_라벨이_두_번이면_멈춘다(self):
        nets = [net('입력 A', '74HC04#5 pin 1'), net('74HC04#5 pin 1', 'LED X 애노드')]
        with self.assertRaisesRegex(SimError, '두 넷'):

    def test_입력값이_없으면_멈춘다(self):

    def test_전원_레일은_상수다(self):

        self.assertEqual(value_of(nets, 'LED X 애노드', {}), 1)

    unittest.main()

```python
# tests/_gate_sim.py
"""결선도 넷에서 74HC 게이트 논리를 돌린다 — v2 시험판의 진리표 검사용.

렌더러 `nets()` 는 "무엇과 무엇이 한 스트립 묶음인가" 까지만 안다. 회로가 교안 함수를
내는지는 아무도 보지 않았다(승인본은 netlist 핀 줄과 대조한다). 시험판은 교안 핀 번호를
일부러 바꾸므로 핀 줄 대조를 못 쓴다 — 그래서 넷을 직접 돌린다. 관측 넷의 **상류만**
평가한다: 중계로 쓴 미사용 게이트는 입력 일부만 물려 있어 출력까지 계산하면 뜬 입력이다.
"""
import re

PINMAP = {
    '74HC04': [((1,), 2), ((3,), 4), ((5,), 6), ((9,), 8), ((11,), 10), ((13,), 12)],
    '74HC20': [((1, 2, 4, 5), 6), ((9, 10, 12, 13), 8)],
    '74HC11': [((1, 2, 13), 12), ((3, 4, 5), 6), ((9, 10, 11), 8)],
}
_FUNC = {'74HC04': lambda v: 1 - v[0],
         '74HC20': lambda v: 0 if all(v) else 1,
         '74HC11': lambda v: 1 if all(v) else 0}
_PIN = re.compile(r'^(74HC\d+)#(\d+) pin (\d+)$')
_INPUT = re.compile(r'^입력 (\S+)$')


class SimError(Exception):
    pass


def _index(nets):
    where = {}
    for i, n in enumerate(nets):
        for lab in n['labels']:
            if lab in where and where[lab] != i:
                raise SimError(f'{lab} 이 두 넷에 있다')
            where[lab] = i
    return where


def _drivers(nets):
    """넷 번호 → [(칩, 행, 입력핀들)] — 그 넷에 출력핀이 앉은 게이트."""
    out = {}
    for i, n in enumerate(nets):
        for lab in n['labels']:
            m = _PIN.match(lab)
            if not m or m.group(1) not in PINMAP:
                continue
            chip, row, pin = m.group(1), m.group(2), int(m.group(3))
            for ins, o in PINMAP[chip]:
                if o == pin:
                    out.setdefault(i, []).append((chip, row, ins))
    return out


def value_of(nets, label, inputs):
    where = _index(nets)
    drivers = _drivers(nets)
    if label not in where:
        raise SimError(f'{label} 이 어느 넷에도 없다')
    memo, busy = {}, set()

    def ev(i):
        if i in memo:
            return memo[i]
        if i in busy:
            raise SimError(f'되먹임: {sorted(nets[i]["labels"])}')
        busy.add(i)
        n = nets[i]
        srcs = []
        if 'vcc' in n['rails']:
            srcs.append(1)
        if 'gnd' in n['rails']:
            srcs.append(0)
        for lab in n['labels']:
            m = _INPUT.match(lab)
            if m:
                if m.group(1) not in inputs:
                    raise SimError(f'{lab} 의 값이 없다')
                srcs.append(int(inputs[m.group(1)]))
        gates = drivers.get(i, [])
        if len(srcs) + len(gates) > 1:
            raise SimError(f'한 넷에 출력 둘 이상: {sorted(n["labels"])}')
        if gates:
            chip, row, ins = gates[0]
            vals = []
            for p in ins:
                lab = f'{chip}#{row} pin {p}'
                if lab not in where:
                    raise SimError(f'뜬 입력: {lab}')
                vals.append(ev(where[lab]))
            srcs.append(_FUNC[chip](vals))
        if not srcs:
            raise SimError(f'뜬 넷: {sorted(n["labels"])}')
        busy.discard(i)
        memo[i] = srcs[0]
        return srcs[0]

    return ev(where[label])
```

주의: 렌더러는 아무 데도 안 물린 칩 핀도 자기 라벨 하나짜리 넷으로 낸다(`['74HC04#7 pin 1']`). 그런 핀이 관측 넷 상류의 게이트 입력이면, 그 넷에는 구동원이 없으므로 `뜬 넷` 으로 멈춘다. 시험 `test_뜬_입력이면_멈춘다` 의 정규식 `'뜬'` 은 `뜬 입력`·`뜬 넷` 둘 다 맞는다.

def _walk(board):
    """사슬 순서대로 (이름, 빌더, 단계, 그릴지) 를 낸다. 막힌 보드 뒤는 기준선이 없으므로
    다음 회로의 첫 보드(`RESTART`, `fresh_board` 로 기준선을 끊는 자리)까지 건너뛴다 —
    MUX 가 막혀도 DMUX 는 잰다."""
    skipping = False
    for name, fn, stage in board.BOARDS:
        if name in board.RESTART:
            skipping = False
        yield name, fn, stage, not skipping
        if name in board.KNOWN_BLOCKERS:
            skipping = True


class Chain(_Trial):
    """단계 사슬 — 막힘은 `KNOWN_BLOCKERS` 로, 새 신호선 수는 `MEASURED_NEW_SIGNAL` 로 고정한다."""

    def test_단계_보드마다_R10_이_통과하거나_기록된_막힘이다(self):
        base = None
        for name, fn, stage, draw in _walk(self.board):
            if name in self.board.RESTART:
                base = None
            if not draw:
                continue
            path = os.path.join(self.tmp, name)
            want = self.board.KNOWN_BLOCKERS.get(name)
            if want:
                with self.assertRaisesRegex(AssertionError, want):
                    fn(path, None, stage=stage, baseline=base)
                continue
            bb = fn(path, None, stage=stage, baseline=base)
            self.assertEqual(bb._block_role_violations(), [], name)
            self.assertEqual(len(bb.new_signal), self.board.MEASURED_NEW_SIGNAL[name], name)
            base = bb.snapshot()

    def test_측정표가_그린_보드를_빠짐없이_덮는다(self):
        drawn = [n for n, _f, _s, draw in _walk(self.board)
                 if draw and n not in self.board.KNOWN_BLOCKERS]
        self.assertEqual(list(self.board.MEASURED_NEW_SIGNAL), drawn)
