# 브레드보드 v2 판형 지원 구현계획

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 가운데 전원 레일이 2열인 v2 판형을 렌더러·손그림 검토·체크리스트 빈 판형·스킬 문서가 지원하게 하고, v1(실험01~04) 산출물은 바이트 하나 바꾸지 않는다.

**Architecture:** 렌더러에 판형 사양 객체 `BoardSpec` 을 두고 본문은 `self.spec` 만 읽는다 — 버전 이름으로 분기하지 않는다. v1 사양은 기존 모듈 상수에서, v2 사양은 `.claude/assets/breadboard_v2/geometry.json` 에서 짓는다. 주차 `배치.py` 의 `BOARD` 는 빌더가 `BreadboardRenderer(..., board=BOARD)` 로 명시적으로 넘기고 `build_all`·손그림 참조가 `check_board()` 로 대조한다. v2 가운데 레일은 두 블록이 같은 x 를 `mid_gnd`·`mid_vcc` 로 부르는 것만으로 한 넷이 된다(넷 단위가 `('rail', x)`).

**Tech Stack:** Python 3 (`.venv`), Pillow, PyMuPDF(`fitz`), pytest + xdist(`-n auto --dist loadgroup`), headless Chrome(판형 수출·PDF).

**Spec:**
- 구조 권위: `docs/plans/2026-10-02-브레드보드-v2-구조-설계.md` (어긋나면 이쪽이 이긴다)

def _load(name="render_breadboard"):
    spec = importlib.util.spec_from_file_location(name, SCRIPT)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod


class V2Template(unittest.TestCase):
    """판형 산출물 — 1배 PNG, 해시, 잭 글자 자리. 판형을 다시 뽑으면 여기서 걸린다."""

def _spec_from_geometry(path):
    """v2 판형 사양을 `geometry.json` 에서 짓는다. 행 수·레일 빈 행·잭 이름이 렌더러가 아는 것과
    다르면 멈춘다 — 판형이 바뀐 것이다. 양식 PNG 는 그 파일 옆에 있다."""
    if not os.path.exists(path):
        raise FileNotFoundError(f'v2 판형 좌표 파일이 없다: {path} — v1 로 넘어가지 않는다')
    with open(path, encoding='utf-8') as f:
        g = json.load(f)
    if g.get('rows') != ROW_MAX or set(g.get('rail_gap_rows', ())) != set(RAIL_GAP_ROWS):
        raise ValueError(f'{path}: 행 수·레일 빈 행이 렌더러({ROW_MAX}행, '
                         f'{sorted(RAIL_GAP_ROWS)})와 다르다')
    cols = {'left': {}, 'right': {}}
    for c in g['columns']:
        side, key = c['name'].split('.')
        if side == 'mid':
            # 가운데 레일은 한 열을 두 블록이 같은 이름으로 부른다 — 넷 단위가 ('rail', x) 라
            # 이것만으로 한 넷이다(설계 §2)
            cols['left'][f'mid_{key}'] = cols['right'][f'mid_{key}'] = c['x']
        else:
            cols[side][key] = c['x']
    for blk in cols.values():
        # 칩 몸통은 e·f 핀 열 안쪽 5px — v1 의 310·349 가 같은 식이다(305+5, 354-5)
        blk['chip_x1'], blk['chip_x2'] = blk['e'] + 5, blk['f'] - 5
    try:
        jacks = {_V2_JACK_KEYS[j['name']]: j['center_x'] for j in g['jacks']}
    except KeyError as e:
        raise ValueError(f'{path}: 모르는 잭 이름 {e} — {sorted(_V2_JACK_KEYS)} 중 하나다') from None
    bottoms = {j['bottom_y'] for j in g['jacks']}
    centers = {j['center_y'] for j in g['jacks']}
    if len(bottoms) != 1 or len(centers) != 1:
        raise ValueError(f'{path}: 잭 높이가 제각각이다 — 잭 선 경로가 하나로 안 정해진다')
    top = g['border']['top']
    return BoardSpec(
        name='v2',
        template=(os.path.join(os.path.dirname(os.path.abspath(path)), g['png']),),
        size=(g['image']['width'], g['image']['height']),

        board_left=g['border']['left'], board_right=g['border']['right'],
        jacks=jacks, jack_bottom=bottoms.pop(),
        # 잭 영역은 보드 위 테두리 위, 잭 선은 테두리 + 8·16 에서 가로지른다 — v1 과 같은 식

        # VCC 와 우 a 열 사이에 둔다
        mid_badge_x=(cols['left']['mid_vcc'] + cols['right']['a']) / 2,

```python
    def _zone(self, end, shared):
        """끝점이 어느 블록 편인가 — 단자대·바깥 레일은 그 블록, 두 블록이 같이 쓰는 레일
        (`shared`, v2 가운데)과 잭·여백은 None."""
        if end[0] != 'hole' or end[1] is None:
            return None
        block, col, _row = end[1]
        return None if self.cols[block][col] in shared else block

    def block_crossings(self):
        """R10 의 지표 — 양 끝이 서로 다른 블록인 배선. 레일↔레일 전원 브리지는 가운데 레일이
        어느 블록도 아니라 여기 안 든다. `wiring()` 에 넣지 않는다 — 도장."""
        shared, out = self._shared_xs(), []
        for _kind, pts, desc in self._segments:
            a, b = self._endpoint(pts[0]), self._endpoint(pts[-1])
            za, zb = self._zone(a, shared), self._zone(b, shared)
            if {za, zb} != {'left', 'right'}:
                continue
            out.append({'ends': {za: a[1], zb: b[1]}, 'desc': desc,
                        'frm': self.addr(*pts[0]), 'to': self.addr(*pts[-1])})
        return out

    def _block_role_violations(self):
        """R10 — LED 전용 블록으로 건너는 선은 칩 핀 스트립 → LED 사슬 신호 스트립뿐이고, 건너는
        선끼리 위아래 순서가 LED 순서와 같다(설계 §5).

        핀 방향표가 없어 '출력' 인지는 보지 않는다(2026-10-03 결정) — LED 에 무엇이 물렸는지는
        R5 와 netlist 대조가 본다. `spec.led_only_blocks` 가 비면(v1) 아무것도 안 낸다.
        """
        led_blocks = self.spec.led_only_blocks
        if not led_blocks:
            return []
        chip_strips = {self._strip(sym) for ch in self.chips
                       if ch['block'] in self.spec.chip_blocks for sym in ch['pins'].values()}
        led_strips = {self._strip(ch['signal']) for ch in self.led_chains
                      if ch['signal'] is not None and ch['signal'][0] in led_blocks}
        bad, ok = [], []
        for w in self.block_crossings():
            led_end = next((s for z, s in w['ends'].items() if z in led_blocks), None)
            chip_end = next((s for z, s in w['ends'].items() if z not in led_blocks), None)
            if (led_end is None or chip_end is None
                    or self._strip(chip_end) not in chip_strips
                    or self._strip(led_end) not in led_strips):
                bad.append(f"R10 블록 건너기 위반: {w['frm']} → {w['to']} · {w['desc']} — "
                           "블록을 건너는 선은 칩 핀 스트립에서 LED 사슬 스트립으로만 간다")
            else:
                ok.append((led_end[2], chip_end[2], w))
        ok.sort(key=lambda t: t[0])
        for (_l1, c1, w1), (_l2, c2, w2) in zip(ok, ok[1:]):
            if c1 > c2:
                bad.append(f"R10 건너는 선 엇갈림: {w1['frm']} → {w1['to']} / "
                           f"{w2['frm']} → {w2['to']} — 칩 쪽 행 순서를 LED 순서와 맞춘다")
        return bad

def _fit_version(board, version, dpi):
    """`version` 판형의 렌더러 좌표표로 맞춘 보정. 검출이 그 판형 열 수로 안 모이면 `GridError`."""

    cols, rows = detect_grid(board.data, version)
    ref_cols = sorted(_reference_cols(mod, version))
    ref_rows = [spec.row0 + (r - 1) * spec.row_pitch for r in range(1, mod.ROW_MAX + 1)]
    ax, bx, res_x = _fit(ref_cols, cols)
    ay, by, res_y = _fit(ref_rows, rows)
    return Calibration(ax, bx, ay, by, res_x, res_y, board.rect,
                       (board.width, board.height), dpi, version)


def _fits(cal):
    return cal.res_x <= MAX_RESIDUAL and cal.res_y <= MAX_RESIDUAL


def _try_fit(board, version, dpi):
    """맞으면 보정, 검출이 안 되거나 잔차가 크면 None."""
    try:
        cal = _fit_version(board, version, dpi)
    except GridError:
        return None
    return cal if _fits(cal) else None


def calibrate(board, version='v1', dpi=300):
    """검출한 격자를 `version` 판형의 렌더러 좌표표에 맞춘다.

    `version` 은 **승인 결선의 판형**이다(`expected_board()`). 그 판형으로 안 맞으면 다른
    판형으로도 맞춰 보되 **진단만 한다** — 맞으면 "판형 버전 불일치" 로 멈추고, 그 판형으로
    계속 가지 않는다. 학생이 다른 배포본 위에 그렸으면 다시 그려야 한다. 검출은 열 수를
    억지로 맞추므로(`_merge_to_count`) 열 수만으로는 판형을 못 가른다.
    """
    mod = load_renderer()
    cal = _try_fit(board, version, dpi)
    if cal is not None:
        return cal
    for other in mod.BOARD_VERSIONS:
        if other != version and _try_fit(board, other, dpi) is not None:
            raise GridError(
                f"판형 버전 불일치 — 승인 결선은 {version} 판형인데 제출 PDF 는 {other} 판형 "
                f"위에 그렸다. {version} 배포본 위에 다시 그려야 한다")
    cal = _fit_version(board, version, dpi)   # 검출 실패면 그 GridError 가 그대로 난다
    raise GridError(
        f"판형이 렌더러 좌표표와 맞지 않는다 "
        f"(가로 잔차 {cal.res_x:.1f}px, 세로 {cal.res_y:.1f}px, 한계 {MAX_RESIDUAL}). "
        "다른 판형 위에 그렸거나 판형이 바뀌었다")
```

    """'좌 b10' -> ('hole', ('left', 'b', 10)). 홀이 아니면 거절한다.

    `board` 는 승인 결선의 판형이다. 가운데 레일을 두 블록이 나눠 쓰는 판형(v2)에서는
    `중앙GND레일9행`·`좌 중앙GND레일9행`·`우 중앙GND레일9행` 이 모두 한 레일

    따로라 블록을 적어야 한다.
    """
    s = (text or "").strip()
    m = _JACK.match(s)
    if m:
        return ('jack', JACK_COLORS[m.group(1)])
    m = _TERMINAL.match(s)
    if m:
        row = int(m.group(3))
        _check_row(row, s)
        return ('hole', (BLOCKS[m.group(1)], m.group(2), row))
    m = _RAIL.match(s)
    if m:
        row = int(m.group(3))
        _check_row(row, s)
        side, col = m.group(1), RAIL_NAMES[m.group(2)]
        if col.startswith('mid_') and board in SHARED_MID_BOARDS:
            return ('hole', ('left', col, row))
        if side is None:
            raise TranscriptError(
                f"{s!r} — {board} 판형에서는 이 레일이 좌·우에 따로 있다. "
                f"'좌 {m.group(2)}{row}행' 처럼 블록을 적는다")
        return ('hole', (BLOCKS[side], col, row))
    raise TranscriptError(
        f"홀 주소로 읽을 수 없다: {s!r} — "
        "'좌 b10' · '좌 GND레일10행' · '전원잭(빨강)' 꼴로 적는다 "
        "(v2 가운데 레일은 '중앙GND레일10행')")

    """주차 `배치.py` 의 `BOARD` — 엔진을 import 하지 않고 ast 로 읽는다. 파일이나 상수가 없으면 v1.

                isinstance(t, ast.Name) and t.id == "BOARD" for t in node.targets):
