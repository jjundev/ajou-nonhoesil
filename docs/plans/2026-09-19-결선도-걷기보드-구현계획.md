```python
        """끝점만 남기면 걷을 선을 그림에 그릴 수 없다 — 걷기 보드가 꼭짓점을 쓴다."""

    def test_removals_는_걷을_배선에_번호와_경로를_준다(self):

        bb = self.mod.BreadboardRenderer(baseline=base)
        bb.enable_steps()
        bb.draw_chip('74HC08', 7)
        rs = bb.removals()
        self.assertEqual([r['n'] for r in rs], list(range(1, len(rs) + 1)))
        self.assertEqual([r['key'] for r in rs], list(base['wires']))   # 기준선 순서

    def test_옛_snapshot_에_paths_가_없으면_points_는_None_이다(self):

        del base['paths']
```

`ENGINE` 의 `snapshot()`·`removed()` 두 메서드를 아래로 바꾼다 (`wiring()` 바로 위).

```python
    def snapshot(self):
        """이 보드의 배선·사슬·칩을 **다음 보드의 기준선**으로 쓸 수 있게 낸다.

        `paths` 는 배선의 꼭짓점이다 — 걷기 보드(`removal_board`)가 뽑을 선을 같은
        경로로 빨간 점선으로 그린다. 끝점만 남기면 그릴 수 없다(2026-09-18 실험03).
        """
        w = self.wiring()
        paths = {}
        for kind, pts, _desc in self._segments:
            paths.setdefault((kind, self._endpoint(pts[0]), self._endpoint(pts[-1])), pts)
        return {
            'wires': {(x['kind'], x['frm'], x['to']): x['desc'] for x in w['wires']},
            'chains': {(ch['anode'], ch['cathode'], ch['res_top'], ch['res_bottom']): ch['tag']
                       for ch in w['chains']},
            'chips': {(ch['block'], ch['pin1_row']): ch['label'] for ch in w['chips']},
            'paths': paths,
        }

    def removals(self):
        """기준선에 있었는데 이번 보드에 없는 **배선** — 걷기 번호를 매겨 낸다.

        순서는 기준선(앞 보드)의 그리기 순서라 결정적이다. netlist 단계 표가 이
        번호(걷기 ①·②)를 인용한다. 칩·사슬은 여기 없고 `removed()` 에만 있다 —
        칩 교체는 순서표에 이미 번호 원이 있다. `points` 는 기준선 `paths` 의 꼭짓점,
        옛 snapshot 이면 None.
        """
        if self.baseline is None:
            return []
        drawn = {(k, self._endpoint(pts[0]), self._endpoint(pts[-1]))
                 for k, pts, _ in self._segments}
        paths = self.baseline.get('paths', {})
        out = []
        for key, desc in self.baseline['wires'].items():
            if key in drawn:
                continue
            out.append({'n': len(out) + 1, 'key': key,
                        'addr': f'{self._sym_addr(key[1])} → {self._sym_addr(key[2])}',
                        'desc': desc, 'points': paths.get(key)})
        return out

    def removed(self):
        """기준선에 있었는데 이번 보드에 없는 것 — 실험실에서 **걷을** 것. 배선 → 칩 →
        사슬 순의 문자열. 배선 줄은 `removals()` 와 같은 순서·같은 수다."""
        if self.baseline is None:
            return []
        out = [f"{r['addr']} · {r['desc']}" for r in self.removals()]
        here = {(c['block'], c['pin1_row']) for c in self.chips}
        for key, label in self.baseline['chips'].items():
            if key not in here:
                out.append(f'칩 {label} ({"좌" if key[0] == "left" else "우"} {key[1]}행) 뽑는다')
        chains = {(ch['anode'], ch['cathode'], ch['res_top'], ch['res_bottom'])
                  for ch in self.led_chains}
        for key, tag in self.baseline['chains'].items():
            if key not in chains:
                out.append(f'LED {tag} 사슬 ({self._sym_addr(("hole", key[0]))}~'
                           f'{self._sym_addr(("hole", key[3]))}) 뽑는다')
        return out
```

```python
    GHOST_KEY = ('sig', ('hole', ('left', 'a', 12)), ('hole', ('left', 'a', 25)))

    def ghost_board(self, drop_wire=False):
        """`first()` 를 기준선으로, 출력 배선 하나만 유령으로 그린 걷기 보드.
        `drop_wire=True` 면 그 배선을 아예 안 그린다 — export 가 잡아야 한다."""
        base = self.first().snapshot()
        ghost = {'wires': {self.GHOST_KEY: 1}, 'title': '걷기 제목', 'subtitle': '걷기 부제'}
        bb = self.mod.BreadboardRenderer(baseline=dict(base, ghost=ghost))
        bb.enable_steps()
        bb.set_header('무시되는 제목', '무시되는 부제')
        c = bb.cols['left']
        bb.draw_chip('74HC08', 7)
        bb.connect_jacks()
        bb.bridge_rails(c['gnd'], c['mid_gnd'], 29, kind='gnd')
        bb.bridge_rails(c['vcc'], c['mid_vcc'], 30, kind='vcc')
        bb.connect_power(7)
        n = None
        if not drop_wire:
            n = bb.step_wire([(c['a'], bb.r_y(12)), (c['a'], bb.r_y(25))], 'sig', '출력 -> L2')
        bb.led_chain(c['a'], 25, 'L2', '최종')
        return bb, n

    def test_ghost_배선은_번호도_점유표도_구간도_받지_않는다(self):
        bb, n = self.ghost_board()
        c = bb.cols['left']
        self.assertIsNone(n)
        self.assertNotIn((c['a'], 12), bb._leads)
        self.assertFalse(any(w['frm'] == self.GHOST_KEY[1] and w['to'] == self.GHOST_KEY[2]
                             for w in bb.wiring()['wires']))
        self.assertEqual(bb.steps, [])                      # 새 선이 없다
        self.assertEqual(bb.ghost_drawn, {self.GHOST_KEY: 1})
        self.assertEqual(len(bb.marker_boxes), 1)           # 걷기 번호 원 하나
        self.assertTrue(bb.partial)

    def test_ghost_모드는_제목을_기준선에서_받고_제출용이_아니다(self):
        bb, _ = self.ghost_board()
        self.assertEqual((bb.title, bb.subtitle, bb.submit), ('걷기 제목', '걷기 부제', False))

    def test_ghost_배선은_removals_와_content_에_번호로_남는다(self):
        bb, _ = self.ghost_board()
        self.assertEqual([(r['n'], r['addr']) for r in bb.removals()], [(1, '좌 a12 → 좌 a25')])
        self.assertEqual(bb.content()['ghost'], [(1, '좌 a12 → 좌 a25')])
        self.assertNotIn('ghost', self.first().content())   # 보통 보드의 도장은 그대로다

    def test_ghost_배선은_PNG_에_빨간_점선으로_찍힌다(self):
        bb, _ = self.ghost_board()
        c = bb.cols['left']
        with tempfile.TemporaryDirectory() as d:
            p = os.path.join(d, 'g.png')
            bb.export(p, max_row=31)
            im = Image.open(p).convert('RGB')
            ys = range(bb.r_y(13), bb.r_y(24))
            self.assertTrue(any(im.getpixel((c['a'], y)) == self.mod.REMOVE[:3] for y in ys))
        plain = self.first()
        with tempfile.TemporaryDirectory() as d:
            p = os.path.join(d, 'p.png')
            plain.partial = True
            plain.export(p, max_row=31)
            im = Image.open(p).convert('RGB')
            self.assertFalse(any(im.getpixel((c['a'], y)) == self.mod.REMOVE[:3] for y in ys))

    def test_ghost_대상을_안_그리면_export_가_잡는다(self):

        with tempfile.TemporaryDirectory() as d:

                bb.export(os.path.join(d, 'x.png'), max_row=31)
```

```python
    def _dashed(self, points, color, width=2, dash=7, gap=5, draw=None):
        d = draw or self.draw
        for (x0, y0), (x1, y1) in zip(points, points[1:]):
            span = ((x1 - x0) ** 2 + (y1 - y0) ** 2) ** 0.5
```

```python
    """두 상자 (x0, y0, x1, y1) 가 겹치는가. 걷기 번호 원 자리 잡기에 쓴다."""
```

```python
    def _marker_removed(self, x, y, n):
        """걷기 번호 — 빨간 채움·흰 숫자. 순서표 원(흰 바탕·색 테두리)과 구분된다."""

    def _ghost_wire(self, points, n):
        """걷기 보드의 뽑을 배선. 유령 레이어에 빨간 점선, overlay 에 걷기 번호 원.

        점유표·구간·레일 접촉에 아무것도 안 남긴다 — 이 선은 이제 없는 것이고, 그 홀에
        다음 회로의 새 선이 앉을 수 있다(실험03 좌 d7·a9). 끝점 원도 안 찍는다.
        번호 원은 가장 긴 구간의 중점에, 기존 원과 겹치면 그 구간을 따라 옮긴다.
        """
        self._dashed(points, REMOVE, width=4, dash=9, gap=6,
                     draw=ImageDraw.Draw(self.ghost_layer))
        if len(points) < 2:
            self._marker_removed(points[0][0], points[0][1], n)
            return
        (x0, y0), (x1, y1) = max(zip(points, points[1:]),
                                 key=lambda s: abs(s[1][0] - s[0][0]) + abs(s[1][1] - s[0][1]))
        span = max(1.0, ((x1 - x0) ** 2 + (y1 - y0) ** 2) ** 0.5)
        ux, uy = (x1 - x0) / span, (y1 - y0) / span
        mx, my = (x0 + x1) / 2, (y0 + y1) / 2
        x, y = mx, my
        for k in (0, 24, -24, 48):
            x, y = mx + ux * k, my + uy * k
            box = (x - 8, y - 8, x + 8, y + 8)
            if not any(_boxes_touch(box, b) for b in self.marker_boxes):
                break
        self._marker_removed(int(x), int(y), n)
```

```python
def removal_board(prev, nxt, title, subtitle):
    """걷기 보드 빌더를 만든다. `prev`·`nxt` 는 `(빌더, stage)`.

    앞 회로(`prev`)를 전부 회색으로 그리고, 다음 회로(`nxt`)에 없는 배선만 빨간 점선 +
    걷기 번호로 그린다. 새 선은 없다. 회로가 바뀌는 경계마다 한 장 — 2026-09-18 실험03
    에서 걷을 두 가닥이 footer 표에만 있어 실험실에서 놓쳤다.

    돌려주는 클로저는 표준 빌더 시그니처라 `BOARDS` 에 그대로 들어간다. `__module__` 이
    엔진이라 주차 파일의 `build_*` 열거(`tests/_boards.py`)에 안 잡힌다 — 걷기 보드는
    `steps` 가 비어 가독성 검사(`steps > 0`)의 대상이 아니다. 주차 파일에 `build_` 로
    시작하는 래퍼를 두지 않는다.

    `baseline` 을 안 넘기면 앞 보드를 임시 렌더해 스스로 만든다 — 단계 래퍼
    (`_eN_board`)가 그렇게 부른다. 사슬(`build_all`)은 앞 보드의 snapshot 을 넘긴다.
    """
    prev_fn, prev_stage = prev
    nxt_fn, nxt_stage = nxt

    def build(output_path, template_path=None, stage=None, baseline=None):
        with tempfile.TemporaryDirectory() as d:
            base = baseline or prev_fn(os.path.join(d, 'prev.png'), template_path,
                                       stage=prev_stage).snapshot()
            nb = nxt_fn(os.path.join(d, 'next.png'), template_path,
                        stage=nxt_stage, baseline=base)
            wires = {r['key']: r['n'] for r in nb.removals()}
        if not wires:
            raise ValueError(f'{title}: 걷을 배선이 없다 — 걷기 보드가 필요 없는 경계다')
        ghost = dict(base, ghost={'wires': wires, 'title': title, 'subtitle': subtitle})
        return prev_fn(output_path, template_path, stage=prev_stage, baseline=ghost)

    build.__name__ = f'removal_{nxt_fn.__name__}'
    return build
```
