# `draft-lab-spec` 실행 검증 구조 구현 계획

> **For agentic workers:** REQUIRED SUB-SKILL: Use
> `subagent-driven-development` (recommended) or `executing-plans` to
> implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for
> tracking.

**Goal:** `draft-lab-spec` 스킬을 **fresh 에이전트로 실행해** 검증하는 구조를
세우고, 지금 판본에 남아 있는 검증 구멍 넷을 닫는다.

**Architecture:** 판정을 사람 눈이 아니라 `compare_specs()` 라는 한 함수에
맡긴다 — 재생성물과 원본을 **계약 필드로만** 대조해 차이를 줄 목록으로 낸다.
그 위에 여섯 가지 실행 시나리오를 런북 문서로 고정해, 스킬을 고칠 때마다
같은 순서로 다시 돌릴 수 있게 한다. 시나리오는 컨트롤러가 fresh 에이전트를
띄워 수행한다 — 구현 에이전트가 띄우면 그 맥락이 새어 통제군이 깨진다.

**Tech Stack:** Python 3.14 표준 라이브러리, `unittest`, git worktree,
Claude Code 서브에이전트, 마크다운

> **머리말의 "태스크마다 서브에이전트" 가 Task 4·5 에는 적용되지 않는다.**
> 그 둘은 컨트롤러가 직접 수행한다 — 안에서 **또 다른** fresh 에이전트를
> 띄워야 하는데, 구현 에이전트가 띄우면 그 맥락이 새어 통제군이 깨진다.
> 게다가 그 둘은 코드 diff 가 아니라 판정을 산출물로 내므로 태스크 리뷰의
> diff 검사가 성립하지 않는다. Task 1·2·3·6 은 평소대로 돌린다.

```python
"""`compare_specs` 가 재생성물의 어긋남을 실제로 짚는지 본다.

격리 재생성은 이 저장소에서 스킬 결함을 찾는 유일한 방법인데, 판정을 사람
눈에 맡기면 매번 기준이 흔들린다. 실제로 한 번은 판정 스크립트가 `C` 열을
가진 표를 전부 훑는 바람에, 레벨 인에이블이라 `1` 이 맞는 표 셋을 FAIL 로
찍었다 — 스크립트가 틀린 것이지 재생성이 틀린 게 아니었다.

여기 있는 테스트는 살아 있는 실험06 명세를 한 군데씩 변형해, 검사기가 그
자리를 짚는지 확인한다. 합성 픽스처를 따로 만들지 않는다.
"""

import tempfile
import unittest
from pathlib import Path

from scripts.lab.specdiff import compare_specs

ROOT = Path(__file__).resolve().parents[1]
WEEK6 = ROOT / "실험06/체크리스트/명세.md"
WEEK12 = ROOT / "실험10/체크리스트/명세.md"


class CompareSpecs(unittest.TestCase):
    def setUp(self):
        self.tmp = tempfile.TemporaryDirectory()
        self.addCleanup(self.tmp.cleanup)

    def _mutated(self, old, new):
        """실험06 명세에서 `old` 를 한 번만 `new` 로 바꾼 사본 경로."""
        text = WEEK6.read_text(encoding="utf-8")
        self.assertIn(old, text, f"픽스처가 바뀌었다 — {old!r} 가 없다")
        target = Path(self.tmp.name) / "명세.md"
        target.write_text(text.replace(old, new, 1), encoding="utf-8")
        return target

    def test_같은_문서끼리는_차이가_없다(self):
        self.assertEqual(compare_specs(WEEK6, WEEK6), [])

    def test_라이징_엣지가_리터럴_1_로_돌아가면_그_자리를_짚는다(self):
        # 이 파일이 존재하는 이유가 이 한 줄이다. 재생성물의 이 셀이 `1` 이면
        # 텍스트 추출의 지뢰를 밟은 것이고, 그것을 사람 눈에 맡기지 않는다.
        target = self._mutated("| 0 | 0->1 | 0 |", "| 0 | 1 | 0 |")
        diffs = compare_specs(WEEK6, target)
        self.assertEqual(len(diffs), 1, diffs)
        self.assertIn("0->1", diffs[0])
        self.assertIn("'1'", diffs[0])

    def test_encoding_이_바뀌면_잡는다(self):
        target = self._mutated("- encoding: 0/1", "- encoding: H/L")
        diffs = compare_specs(WEEK6, target)
        self.assertTrue(any("encoding" in d for d in diffs), diffs)

    def test_source_가_바뀌면_잡는다(self):
        target = self._mutated("- source: manual:p9", "- source: manual:p99")
        diffs = compare_specs(WEEK6, target)
        self.assertTrue(any("source" in d for d in diffs), diffs)

    def test_전원_레일이_빠지면_잡는다(self):
        target = self._mutated('power: {VCC: "+5V", GND: "GND"}',
                               'power: {VCC: "+5V"}')
        diffs = compare_specs(WEEK6, target)
        self.assertTrue(any("전원" in d for d in diffs), diffs)

    def test_산문이_달라도_차이로_보지_않는다(self):
        # 재생성물이 같은 사실을 다른 문장으로 쓰는 것은 결함이 아니다.
        target = self._mutated("교안 p7. `74HC00` 쿼드 NAND",
                               "교안 p7 의 회로다. `74HC00` 쿼드 NAND")
        self.assertEqual(compare_specs(WEEK6, target), [])

    def test_다른_주차끼리는_양쪽_방향을_다_보고한다(self):
        # 실험06 은 표 T1~T5·항목 다섯, 실험10 은 표 T1~T2·항목 둘이다.
        forward = compare_specs(WEEK6, WEEK12)
        self.assertTrue(any("원본에만 있다" in d for d in forward), forward)
        backward = compare_specs(WEEK12, WEEK6)
        self.assertTrue(any("재생성에만 있다" in d for d in backward), backward)
```

`scripts/lab/specdiff.py` 를 새로 만든다:

```python
"""재생성물과 원본을 **계약 필드로만** 대조한다.

`validate.py` 가 한 문서의 내부 정합을 보는 것과 달리, 여기는 **두 문서를**
본다. 격리 재생성의 합격 판정이 사람 눈에 걸려 있으면 매번 기준이 흔들리기
때문이다.

**산문은 보지 않는다.** 제목·`expected`·본문 서술은 같은 사실을 다르게 써도
결함이 아니다. 보는 것은 소비자가 실제로 기대는 필드뿐이다 — 프론트매터의
`week`·`experiment`·`power`, 항목의 `device`·`ics`·`tables`, 표의 `kind`·
`encoding`·`source`·`role` 과 셀 전부.
"""

from .spec_parser import parse_spec

# 대조 대상. **여기 없는 필드는 달라도 결함이 아니다.**
_ITEM_FIELDS = ("device", "ics", "tables")
_TABLE_FIELDS = ("kind", "encoding", "source", "role")


def compare_specs(original, regenerated) -> list:
    """두 명세의 계약 필드를 대조해 차이를 줄 목록으로 돌려준다.

    같으면 빈 리스트다. 파싱조차 안 되면 대조 이전의 문제이므로
    `SpecError` 를 그대로 올려보낸다 — 여기서 삼키면 "차이 없음" 과
    "읽지도 못함" 이 구별되지 않는다.
    """
    a, b = parse_spec(original), parse_spec(regenerated)
    out = []

    for field in ("week", "experiment"):
        x, y = getattr(a, field), getattr(b, field)
        if x != y:
            out.append(f"프론트매터 {field}: 원본 {x!r} / 재생성 {y!r}")

    if sorted(a.power) != sorted(b.power):
        out.append(f"전원 레일 이름: 원본 {sorted(a.power)} / "
                   f"재생성 {sorted(b.power)}")
    else:
        for rail in sorted(a.power):
            if a.power[rail] != b.power[rail]:
                out.append(f"전원 {rail}: 원본 {a.power[rail]!r} / "
                           f"재생성 {b.power[rail]!r}")

    out += _by_id("항목", {i.id: i for i in a.items},
                  {i.id: i for i in b.items}, _ITEM_FIELDS)
    out += _by_id("표", {t.id: t for t in a.tables},
                  {t.id: t for t in b.tables}, _TABLE_FIELDS)

    shared = sorted({t.id for t in a.tables} & {t.id for t in b.tables})
    for tid in shared:
        out += _cells(tid, a.table(tid), b.table(tid))
    return out


def _by_id(label, av, bv, fields) -> list:
    """ID 로 짝지어 대조한다. 한쪽에만 있는 것도 보고한다."""
    out = [f"{label} {k}: 원본에만 있다" for k in sorted(set(av) - set(bv))]
    out += [f"{label} {k}: 재생성에만 있다" for k in sorted(set(bv) - set(av))]
    for key in sorted(set(av) & set(bv)):
        for field in fields:
            x, y = getattr(av[key], field), getattr(bv[key], field)
            if x != y:
                out.append(f"{label} {key} {field}: 원본 {x!r} / 재생성 {y!r}")
    return out


def _cells(tid, a, b) -> list:
    """셀 대조. 헤더나 행 수가 다르면 그것만 보고하고 셀은 안 본다 —
    어긋난 표에서 셀을 짝지으면 어긋난 자리마다 줄이 나와 목록이 못 읽게 된다.
    """
    if a.header != b.header:
        return [f"표 {tid} 헤더: 원본 {a.header} / 재생성 {b.header}"]
    if len(a.rows) != len(b.rows):
        return [f"표 {tid} 행 수: 원본 {len(a.rows)}행 / "
                f"재생성 {len(b.rows)}행"]
    out = []
    for n, (ra, rb) in enumerate(zip(a.rows, b.rows), start=1):
        if len(ra) != len(rb):
            out.append(f"표 {tid} 행{n} 셀 수: 원본 {len(ra)} / 재생성 {len(rb)}")
            continue
        for i, (ca, cb) in enumerate(zip(ra, rb)):
            if ca != cb:
                col = a.header[i] if i < len(a.header) else f"열{i + 1}"
                out.append(f"표 {tid} 행{n} {col}: 원본 {ca!r} / 재생성 {cb!r}")
    return out
```
