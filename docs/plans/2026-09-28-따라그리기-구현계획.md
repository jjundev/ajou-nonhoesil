def _load(name="render_breadboard"):
    spec = importlib.util.spec_from_file_location(name, SCRIPT)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)

class Layer(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.mod = _load()

    def _jacks(self, layer=None):
        """전원 잭 두 가닥 — 1번 +5V, 2번 0V."""
        bb = self.mod.BreadboardRenderer(None, layer=layer)
        bb.enable_steps()
        bb.set_header('층', '테스트')
        bb.connect_jacks()
        return bb

    def _px(self, bb, n):
        """n 번 배선 첫 구간 가운데의 overlay 픽셀."""
        _kind, pts, _desc = bb._segments[n - 1]
        (x0, y0), (x1, y1) = pts[0], pts[1]
        s = self.mod.SCALE
        return bb.overlay.getpixel(((x0 + x1) // 2 * s, (y0 + y1) // 2 * s))

        bb = self._jacks()
        self.assertIsNone(bb.layer)
        self.assertEqual(self._px(bb, 1), self.mod.WIRE['vcc'])
        self.assertEqual(self._px(bb, 2), self.mod.WIRE['gnd'])

        self.assertEqual(self._px(bb, 2)[3], 0)
        self.assertEqual(len(bb.marker_boxes), 1)

    def test_번호는_완성본_그대로(self):
        full, part = self._jacks(), self._jacks((2, 2))
        self.assertEqual([s['n'] for s in part.steps], [1, 2])
        self.assertEqual(part.sidecar()['steps'], full.sidecar()['steps'])

    def test_뒤_배선도_점유표에_남는다(self):
        full, part = self._jacks(), self._jacks((1, 1))
        self.assertEqual(set(part._leads), set(full._leads))
        self.assertEqual(part._passed, full._passed)

    def test_잘못된_층은_거절한다(self):

        self.assertNotIn('layer', self._jacks().content())
        self.assertEqual(self._jacks((1, 2)).content()['layer'], [1, 2])

    unittest.main()

```python
    def step_wire(self, points, kind, desc, width=3, marker_at=None, count=1,
                  frm=None, to=None):
        """배선 하나 + 결선 순서표 한 줄.

        `desc` 는 짧은 뜻풀이만 준다 — 어디서 어디로 가는지는 좌표에서
        자동으로 뽑는다 ('좌 b7 -> 좌 GND레일7행').

        기준선(`baseline`)에 같은 (색, 끝점, 끝점) 이 있으면 **회색으로 그리고
        번호를 주지 않는다** — 앞 단계에서 꽂아 둔 그대로다. 점유표·구간·레일
        접촉은 똑같이 남긴다: 회색이어도 그림대로 꽂히는지는 verify() 가 본다.

        따라 그리기 층(`layer`)이면 번호로 모양을 정한다(`_layer_view`). 어느 모양이든
        점유표·구간·결선 순서표에는 남는다 — 번호와 뒤 배선 자리가 완성본과 같아야 한다.
        """
        key = (kind, self._endpoint(points[0]), self._endpoint(points[-1]))
        if self.ghost is not None and key in self.ghost['wires']:
            n = self.ghost['wires'][key]
            self._ghost_wire(points, n)
            self.ghost_drawn[key] = n
            return None
        kept = self.baseline is not None and key in self.baseline['wires']
        view = self._layer_view(len(self.steps) + 1) if self.numbered and not kept else 'shown'
        draw = self._hidden_draw if view == 'hidden' else self.draw
        color = KEEP if kept or view == 'faded' else WIRE[kind]
        draw.line(points, fill=color, width=width, joint='curve')
        for p in (points[0], points[-1]):
            draw.ellipse([p[0]-3, p[1]-3, p[0]+3, p[1]+3], fill=color)
        for p in (points[0], points[-1]):
            self.lead(p[0], p[1], f'{KIND_KR[kind]} 배선 · {desc}')
            self.touch_rail(kind, p[0])
        self._segments.append((kind, [tuple(p) for p in points], desc))
        self._register_pass(points)
        if not kept and kind in ('sig', 'mid'):
            self.new_signal.append(key)
        if kept:
            self.kept.append(desc)
            return None
        if not self.numbered:
            return None

        n = len(self.steps) + 1
        if view == 'shown':
            if marker_at is None:
                (x0, y0), (x1, y1) = points[0], points[1] if len(points) > 1 else points[0]
                marker_at = ((x0 + x1) // 2, (y0 + y1) // 2)
            self._marker(marker_at[0], marker_at[1], n, color)

        frm = frm or self.addr(*points[0])
        to = to or self.addr(*points[-1])
        self.steps.append({'n': n, 'kind': kind, 'frm': frm, 'to': to,
                           'desc': desc, 'count': count})
        return n
```

class LayerError(ValueError):
    """`LAYERS` 가 없거나 완성본 번호를 빈틈없이 덮지 않는다."""


@dataclass
class Page:
    image: str        # 작업 폴더 안 PNG 파일 이름
    heading: str
    rows: list        # 사이드카 steps 중 이 쪽 번호
    notes: list       # 완성본 notes — 0쪽에만
    legend: bool      # 층 쪽 안내 한 줄


def check_layers(name, layers, n_steps):
    """층 경계가 1..n_steps 를 빈틈·겹침 없이 덮는지. 어긋나면 번호를 말하고 멈춘다."""
    expected = 1
    for lo, hi, label in layers:
        if lo != expected:
            raise LayerError(f'{name}: 층 "{label}" 은 {expected}번에서 시작해야 하는데 {lo}번에서 시작한다')
        if hi < lo:
            raise LayerError(f'{name}: 층 "{label}" 의 끝 {hi}번이 시작 {lo}번보다 앞이다')
        expected = hi + 1
    if expected - 1 != n_steps:
        raise LayerError(f'{name}: 층이 {expected - 1}번에서 끝나는데 완성본 배선은 {n_steps}개다')


def build_pages(board, work_dir):
    """완성본마다 0쪽 + 층 쪽. 그림과 사이드카는 `work_dir` 에 쓴다."""
    work_dir = Path(work_dir)
    layers = getattr(board, 'LAYERS', None)
    if not layers:
        raise LayerError('배치.py 에 LAYERS 가 없다 — 완성본마다 (lo, hi, 이름) 층 목록을 둔다')
    pages = []
    for filename, fn in board.FINALS:
        if filename not in layers:
            raise LayerError(f'{filename}: LAYERS 에 층 목록이 없다')
        final = fn(str(work_dir / filename), None)
        steps = final.sidecar()['steps']
        check_layers(filename, layers[filename], len(steps))
        pages.append(Page(filename, f'{final.title} — 완성본 (목표 그림)', [],
                          list(final.notes), False))
        stem, total = Path(filename).stem, len(layers[filename])
        for i, (lo, hi, label) in enumerate(layers[filename], 1):
            name = f'{stem}_층{i}.png'
            fn(str(work_dir / name), None, layer=(lo, hi))
            rows = [s for s in steps if lo <= s['n'] <= hi]
            pages.append(Page(name, f'{final.title} — {i}/{total} · {label} ({lo}~{hi}번)',
                              rows, [], True))
    return pages


def _row(s):
    return (f'<tr><td class="n kind-{s["kind"]}">{s["n"]}</td>'
            f'<td>{html.escape(s["from"])} → {html.escape(s["to"])}</td>'
            f'<td>{html.escape(s["desc"])}</td></tr>')


def render_html(pages, title):
    body = []
    for p in pages:
        parts = [f'<section class="page"><h1>{html.escape(p.heading)}</h1>']
        if p.legend:
            parts.append(f'<p class="legend">{html.escape(LEGEND)}</p>')
        parts.append(f'<img src="{quote(p.image)}" alt="{html.escape(p.heading)}">')
        if p.rows:
            parts.append('<table><thead><tr><th>번호</th><th>홀</th><th>뜻</th></tr></thead><tbody>'
                         + ''.join(_row(s) for s in p.rows) + '</tbody></table>')
        if p.notes:
            parts.append('<ul class="notes">'
                         + ''.join(f'<li>{html.escape(t)}</li>' for t in p.notes) + '</ul>')
        parts.append('</section>')
        body.append(''.join(parts))
    return ('<!doctype html><html lang="ko"><head><meta charset="utf-8">'
            f'<title>{html.escape(title)}</title><style>{CSS}</style></head><body>\n'
            + '\n'.join(body) + '\n</body></html>\n')


def make_pdf(board, out):
    """`board` 의 따라 그리기 PDF 를 `out` 에 쓴다. 쪽 수(섹션 수)를 돌려준다."""
    out = Path(out)
    with tempfile.TemporaryDirectory() as d:
        pages = build_pages(board, d)
        html_path = Path(d) / 'guide.html'
        html_path.write_text(render_html(pages, out.stem), encoding='utf-8')
        html_to_pdf(html_path, out)
    return len(pages)


def output_path(experiment):
    folder = config.experiment_folder_name(experiment)
    return (REPO_ROOT / config.experiment_dir(experiment) / '결선도' / OUT_DIR
            / f'{folder}_따라그리기.pdf')


def make_guide(experiment, out=None):
    board = load_renderer().load_board(experiment)
    out = Path(out) if out else output_path(experiment)
    return out, make_pdf(board, out)


def main(argv=None):
    parser = argparse.ArgumentParser(description='완성본 결선도를 층으로 나눈 따라 그리기 PDF')
    parser.add_argument('experiment', type=int, help='실험 번호 (배치.py 에 LAYERS 가 있어야 한다)')
    parser.add_argument('--out', default=None, help='PDF 경로 (기본: 결선도/팀원전달용/실험NN_따라그리기.pdf)')
    args = parser.parse_args(argv)
    try:
        out, n = make_guide(args.experiment, args.out)
    except (LayerError, PdfError, FileNotFoundError) as e:
        print(f'[ERROR] {e}', file=sys.stderr)
        return 1
    print(f'[SUCCESS] {out} ({n}쪽)')
    return 0


if __name__ == '__main__':
    sys.exit(main())
